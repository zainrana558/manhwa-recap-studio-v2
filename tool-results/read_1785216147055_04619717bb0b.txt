    96→    96→// --- MangaHere (mangahere.cc) ---

    98→    98→const MANGAHERE_BASE = 'https://www.mangahere.cc'
    99→    99→const MANGAHERE_CDN = 'https://zjcdn.mangahere.org'
   100→   100→
   101→   101→/**
   102→   102→ * Fetch the chapter list for a manga from MangaHere.
   103→   103→ * mangaId is the MangaHere slug (e.g. "solo_leveling").
   104→   104→ * Returns chapters oldest-first.
   105→   105→ */
   106→   106→export async function fetchMangaHereChapters(
   107→   107→  mangaSlug: string,
   108→   108→  chapterLimit: number,
   109→   109→): Promise<
   110→   110→  Array<{
   111→   111→    mangadexId: string // chapter slug e.g. "c001" (kept as mangadexId for DB compat)
   112→   112→    chapterNum: string | null
   113→   113→    title: string | null
   114→   114→    language: string
   115→   115→    pageCount: number
   116→   116→    external: boolean
   117→   117→  }>
   118→   118→> {
   119→   119→  const url = `${MANGAHERE_BASE}/manga/${mangaSlug}/`
   120→   120→  const res = await fetch(url, {
   121→   121→    headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36' },
   122→   122→  })
   123→   123→  if (!res.ok) {
   124→   124→    throw new Error(`MangaHere chapters ${res.status} for ${mangaSlug}`)
   125→   125→  }
   126→   126→  const html = await res.text()
   127→   127→
   128→   128→  // Parse chapter links: href="/manga/{slug}/c{chapter}/1.html"
   129→   129→  const chapters: Array<{
   130→   130→    mangadexId: string
   131→   131→    chapterNum: string | null
   132→   132→    title: string | null
   133→   133→    language: string
   134→   134→    pageCount: number
   135→   135→    external: boolean
   136→   136→  }> = []
   137→   137→  const seen = new Set<string>()
   138→   138→  const chapterRegex = new RegExp(
   139→   139→    `href="/manga/${mangaSlug}/((?:v\\d+/)?c[0-9.]+)/1\\.html"`,
   140→   140→    'gi',
   141→   141→  )
   142→   142→  let match
   143→   143→  while ((match = chapterRegex.exec(html)) !== null) {
   144→   144→    const chapSlug = match[1]
   145→   145→    if (seen.has(chapSlug)) continue
   146→   146→    seen.add(chapSlug)
   147→   147→    const chapterNum = chapSlug.match(/c([0-9.]+)/) ? (chapSlug.match(/c([0-9.]+)/)![1].replace(/^0+/, '') || '0') : '0'
   148→   148→    chapters.push({
   149→   149→      mangadexId: chapSlug,
   150→   150→      chapterNum,
   151→   151→      title: null,
   152→   152→      language: 'en',
   153→   153→      pageCount: 0,
   154→   154→      external: false,
   155→   155→    })
   156→   156→  }
   157→   157→
   158→   158→  // MangaHere returns newest-first; reverse to oldest-first.
   159→   159→  chapters.reverse()
   160→   160→
   161→   161→  // Apply chapter limit (0 = all).
   162→   162→  const limited = chapterLimit > 0 ? chapters.slice(0, chapterLimit) : chapters
   163→   163→  return limited
   164→   164→}

   166→   166→/**
   167→   167→ * Extract all image URLs for a MangaHere chapter by scraping the chapter page HTML.
   168→   168→ *
   169→   169→ * MangaHere loads images via obfuscated JavaScript. The image filenames are
   170→   170→ * embedded in the HTML as pipe-separated values. We extract them and construct
   171→   171→ * full CDN URLs.
   172→   172→ */
   173→   173→export async function fetchMangaHereChapterImages(
   174→   174→  mangaSlug: string,
   175→   175→  chapterSlug: string,
   176→   176→): Promise<string[]> {
   177→   177→  const url = `${MANGAHERE_BASE}/manga/${mangaSlug}/${chapterSlug}/1.html`
   178→   178→  const res = await fetch(url, {
   179→   179→    headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36' },
   180→   180→  })
   181→   181→  if (!res.ok) {
   182→   182→    throw new Error(`MangaHere chapter ${res.status} for ${mangaSlug}/${chapterSlug}`)
   183→   183→  }
   184→   184→  const html = await res.text()
   185→   185→
   186→   186→  // Extract store ID from cover image URL: store/manga/{storeId}/cover.jpg
   187→   187→  const storeMatch = html.match(/store\/manga\/(\d+)/)
   188→   188→  const storeId = storeMatch?.[1]
   189→   189→  if (!storeId) {
   190→   190→    throw new Error(`Could not extract store ID from ${url}`)
   191→   191→  }
   192→   192→
   193→   193→  // Extract chapter folder from chapter slug: "v72/c700" → "700", "c001" → "001", "c200.5" → "200.5"
   194→   194→  const chapterMatch = chapterSlug.match(/c([0-9.]+)/)
   195→   195→  const chapterFolder = chapterMatch
   196→   196→    ? chapterMatch[1].padStart(3, '0')
   197→   197→    : chapterSlug.replace(/^c/, '').padStart(3, '0')
   198→   198→
   199→   199→  // Extract image filenames from obfuscated JavaScript.
   200→   200→  // Pattern: {letter}{date}_{time}_{number} e.g. h20181105_144325_927
   201→   201→  const filenameRegex = /([a-z]\d{8}_\d{6}_[a-z0-9]+)/gi
   202→   202→  const filenames = new Set<string>()
   203→   203→  let m
   204→   204→  while ((m = filenameRegex.exec(html)) !== null) {
   205→   205→    filenames.add(m[1])
   206→   206→  }
   207→   207→
   208→   208→  if (filenames.size === 0) {
   209→   209→    throw new Error(`No image filenames found in ${url}`)
   210→   210→  }
   211→   211→
   212→   212→  // Construct full CDN URLs
   213→   213→  const imageUrls = Array.from(filenames).map(
   214→   214→    (fn) =>
   215→   215→      `${MANGAHERE_CDN}/store/manga/${storeId}/${chapterFolder}.0/compressed/${fn}.jpg`,
   216→   216→  )
   217→   217→
   218→   218→  return imageUrls
   219→   219→}

   222→   222→ * Download a single MangaHere image to disk.
   223→   223→ * CRITICAL: MangaHere CDN requires `Referer: https://www.mangahere.cc/` header.
   224→   224→ */
   225→   225→export async function downloadMangaHereImage(
   226→   226→  imageUrl: string,
   227→   227→  destPath: string,

   233→   233→      Accept: 'image/*',
   234→   234→    },
   235→   235→  })

   237→   237→    throw new Error(`MangaHere image download ${res.status}: ${imageUrl}`)
   238→   238→  }
   239→   239→  const buf = Buffer.from(await res.arrayBuffer())
   240→   240→  await fs.writeFile(destPath, buf)
   241→   241→}

   447→   447→

   449→   449→    `href="([^"]*episode_no=${episodeNo})"`,
   450→   450→    'i',
   451→   451→  )
   452→   452→  const viewerMatch = listHtml.match(viewerRegex)
   453→   453→  if (!viewerMatch) {
   454→   454→    throw new Error(`Episode ${episodeNo} not found for title_no=${titleNo}`)
   455→   455→  }
   456→   456→  const viewerUrl = viewerMatch[1].startsWith('http')
   457→   457→    ? viewerMatch[1]
   458→   458→    : `${WEBTOONS_BASE}${viewerMatch[1]}`
   459→   459→
   460→   460→  const res = await fetch(viewerUrl, {
   461→   461→    headers: {
   462→   462→      'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
   463→   463→      'Accept-Language': 'en-US,en;q=0.9',
   464→   464→      Referer: `${WEBTOONS_BASE}/`,
   465→   465→    },
   466→   466→  })
   467→   467→  if (!res.ok) {
   468→   468→    throw new Error(`Webtoons viewer ${res.status}`)
   469→   469→  }
   470→   470→  const html = await res.text()
   471→   471→
   472→   472→  // Webtoons embeds image URLs in data-url attributes.
   473→   473→  const images: string[] = []
   474→   474→  const dataUrlRegex = /data-url="(https:\/\/webtoon-phinf\.pstatic\.net\/[^"]+)"/gi
   475→   475→  let m
   476→   476→  while ((m = dataUrlRegex.exec(html)) !== null) {
   477→   477→    images.push(m[1])
   478→   478→  }
   479→   479→
   480→   480→  // Fallback: try src attributes.
   481→   481→  if (images.length === 0) {
   482→   482→    const srcRegex = /src="(https:\/\/webtoon-phinf\.pstatic\.net\/[^"]+)"/gi
   483→   483→    while ((m = srcRegex.exec(html)) !== null) {
   484→   484→      images.push(m[1])
   485→   485→    }
   486→   486→  }
   487→   487→
   488→   488→  return images
   489→   489→}

   565→   565→let zaiPromise: Promise<unknown> | null = null

   567→   567→async function getZai() {
   568→   568→  // Lazy-load so the service can boot even if the SDK has issues at first run.
   569→   569→  if (!zaiPromise) {
   570→   570→    zaiPromise = (async () => {
   571→   571→      const ZAI = (await import('z-ai-web-dev-sdk')).default
   572→   572→      return await ZAI.create()
   573→   573→    })()
   574→   574→  }
   575→   575→  return await zaiPromise
   576→   576→}

   578→   578→/**
   579→   579→ * Generate an English narrative summary for a chapter by sampling up to 9
   580→   580→ * images (first 3, middle 3, last 3) and asking the VLM to describe them.
   581→   581→ *
   582→   582→ * Falls back to a minimal summary on any error so the pipeline never blocks.
   583→   583→ * NOTE: This is the OLD chapter-level summary. Prefer generateImageNarrations
   584→   584→ * for per-image narration that stays in sync with the video frames.
   585→   585→ */
   586→   586→export async function generateChapterSummary(
   587→   587→  imagePaths: string[],
   588→   588→): Promise<string> {
   589→   589→  if (imagePaths.length === 0) {
   590→   590→    return 'The chapter continues the story.'
   591→   591→  }
   592→   592→
   593→   593→  // Pick up to 9 sample images: first 3, middle 3, last 3.
   594→   594→  const sample = pickSampleImages(imagePaths, 9)
   595→   595→
   596→   596→  try {
   597→   597→    const zai = await getZai()
   598→   598→
   599→   599→    // Build content array: text prompt + each sample image as a base64 data URL.
   600→   600→    const content: Array<
   601→   601→      | { type: 'text'; text: string }
   602→   602→      | { type: 'image_url'; image_url: { url: string } }
   603→   603→    > = [
   604→   604→      {
   605→   605→        type: 'text',
   606→   606→        text:
   607→   607→          'You are summarizing a manhwa/manga chapter for a recap video. ' +
   608→   608→          'Look at these panel images (ordered from the beginning, middle, and end of the chapter) ' +
   609→   609→          'and write a detailed ENGLISH narrative summary of what happens: the events, ' +
   610→   610→          'character actions, key dialogue, and emotional beats. ' +
   611→   611→          'Write 3 to 6 sentences in third person, present tense, as if narrating a story. ' +
   612→   612→          'Do not mention chapter numbers. Do not mention that you are looking at images. ' +
   613→   613→          'Output only the summary text.',
   614→   614→      },
   615→   615→    ]
   616→   616→
   617→   617→    for (const p of sample) {
   618→   618→      const buf = await fs.readFile(p)
   619→   619→      const b64 = buf.toString('base64')
   620→   620→      const mime = p.toLowerCase().endsWith('.png') ? 'image/png' : 'image/jpeg'
   621→   621→      content.push({
   622→   622→        type: 'image_url',
   623→   623→        image_url: { url: `data:${mime};base64,${b64}` },
   624→   624→      })
   625→   625→    }
   626→   626→
   627→   627→    const zaiAny = zai as {
   628→   628→      chat: {
   629→   629→        completions: {
   630→   630→          createVision: (opts: {
   631→   631→            messages: Array<{ role: string; content: typeof content }>
   632→   632→            thinking: { type: string }
   633→   633→          }) => Promise<{
   634→   634→            choices?: Array<{ message?: { content?: string } }>
   635→   635→          }>
   636→   636→        }
   637→   637→      }
   638→   638→    }
   639→   639→
   640→   640→    const resp = await zaiAny.chat.completions.createVision({
   641→   641→      messages: [{ role: 'user', content }],
   642→   642→      thinking: { type: 'disabled' },
   643→   643→    })
   644→   644→
   645→   645→    const text = resp?.choices?.[0]?.message?.content?.trim()
   646→   646→    if (text && text.length > 0) {
   647→   647→      return text
   648→   648→    }
   649→   649→    return 'The chapter continues the story.'
   650→   650→  } catch (err) {
   651→   651→    console.error('[VLM] summary generation failed:', err)
   652→   652→    return 'The chapter continues the story.'
   653→   653→  }
   654→   654→}

   656→   656→/**
   657→   657→ * Generate per-image narrations: send each image to the VLM individually and
   658→   658→ * get 2-4 sentences of narration describing exactly what's in that image.
   659→   659→ * This produces perfect sync — when the video shows image N, the narration
   660→   660→ * describes image N.
   661→   661→ *
   662→   662→ * Processes images with limited concurrency (3 at a time) to balance speed
   663→   663→ * and API load. Falls back to a generic sentence per image on any error so
   664→   664→ * the pipeline never blocks.
   665→   665→ *
   666→   666→ * Returns an array of { image, text } in the same order as imagePaths.
   667→   667→ */
   668→   668→export async function generateImageNarrations(
   669→   669→  imagePaths: string[],
   670→   670→  onProgress?: (done: number, total: number) => void,
   671→   671→): Promise<Array<{ image: string; text: string }>> {
   672→   672→  if (imagePaths.length === 0) {
   673→   673→    return []
   674→   674→  }
   675→   675→
   676→   676→  const results: Array<{ image: string; text: string }> = new Array(imagePaths.length)
   677→   677→  const CONCURRENCY = 3
   678→   678→
   679→   679→  // Process in batches of CONCURRENCY.
   680→   680→  for (let i = 0; i < imagePaths.length; i += CONCURRENCY) {
   681→   681→    const batch = imagePaths.slice(i, i + CONCURRENCY)
   682→   682→    const texts = await Promise.all(
   683→   683→      batch.map(async (imgPath, batchIdx) => {
   684→   684→        const globalIdx = i + batchIdx
   685→   685→        try {
   686→   686→          const text = await narrateSingleImage(imgPath)
   687→   687→          return text
   688→   688→        } catch (err) {
   689→   689→          console.error(`[VLM] image ${globalIdx + 1} narration failed:`, err)
   690→   690→          return 'The scene continues to unfold.'
   691→   691→        }
   692→   692→      }),
   693→   693→    )
   694→   694→    for (let j = 0; j < batch.length; j++) {
   695→   695→      const globalIdx = i + j
   696→   696→      results[globalIdx] = {
   697→   697→        image: path.basename(imagePaths[globalIdx]),
   698→   698→        text: texts[j],
   699→   699→      }
   700→   700→    }
   701→   701→    if (onProgress) {
   702→   702→      onProgress(Math.min(i + CONCURRENCY, imagePaths.length), imagePaths.length)
   703→   703→    }
   704→   704→  }
   705→   705→
   706→   706→  return results
   707→   707→}

   782→   782→/**
   783→   783→ * Pick up to `maxCount` sample images from a sorted list: first N, middle N, last N.
   784→   784→ */
   785→   785→function pickSampleImages(paths: string[], maxCount: number): string[] {
   786→   786→  if (paths.length <= maxCount) return paths.slice()
   787→   787→  const n = Math.floor(maxCount / 3)
   788→   788→  const first = paths.slice(0, n)
   789→   789→  const midStart = Math.floor(paths.length / 2) - Math.floor(n / 2)
   790→   790→  const middle = paths.slice(midStart, midStart + n)
   791→   791→  const last = paths.slice(paths.length - n)
   792→   792→  // Dedupe in case of overlap on small arrays.
   793→   793→  const seen = new Set<string>()
   794→   794→  const out: string[] = []
   795→   795→  for (const p of [...first, ...middle, ...last]) {
   796→   796→    if (!seen.has(p)) {
   797→   797→      seen.add(p)
   798→   798→      out.push(p)
   799→   799→    }
   800→   800→  }
   801→   801→  return out.slice(0, maxCount)
   802→   802→}
