    54→    54→// HTTP server + socket.io
    55→    55→//
    56→    56→// IMPORTANT: socket.io is configured with `path: '/'` (required by the Caddy
    57→    57→// gateway — the frontend connects via `io("/?XTransformPort=3001")`). With
    58→    58→// path `/`, engine.io's attach() wrapper claims ALL HTTP requests, including
    59→    59→// our `/internal/*` endpoints. To work around this, we install an engine.io
    60→    60→// middleware that intercepts `/internal/*` requests and routes them to our
    61→    61→// HTTP handler before engine.io processes them.
    62→    62→// ---------------------------------------------------------------------------

    64→    64→async function httpHandler(req: IncomingMessage, res: ServerResponse) {
    65→    65→  // CORS pre-flight
    66→    66→  res.setHeader('Access-Control-Allow-Origin', '*')
    67→    67→  res.setHeader('Access-Control-Allow-Methods', 'GET, POST, OPTIONS')
    68→    68→  res.setHeader('Access-Control-Allow-Headers', 'Content-Type')
    69→    69→  if (req.method === 'OPTIONS') {
    70→    70→    res.writeHead(204)
    71→    71→    res.end()
    72→    72→    return
    73→    73→  }
    74→    74→
    75→    75→  const url = (req.url || '/').split('?')[0]
    76→    76→
    77→    77→  // Collect request body for POSTs.
    78→    78→  const body = await readBody(req)
    79→    79→
    80→    80→  if (req.method === 'GET' && url === '/internal/health') {
    81→    81→    sendJson(res, 200, { ok: true, port: PORT, queue: queueState() })
    82→    82→    return
    83→    83→  }
    84→    84→
    85→    85→  if (req.method === 'POST' && url === '/internal/start') {
    86→    86→    const { jobId } = body as { jobId?: string }
    87→    87→    if (!jobId) {
    88→    88→      sendJson(res, 400, { error: 'jobId required' })
    89→    89→      return
    90→    90→    }
    91→    91→    const job = await db.job.findUnique({ where: { id: jobId } })
    92→    92→    if (!job) {
    93→    93→      sendJson(res, 404, { error: 'job not found' })
    94→    94→      return
    95→    95→    }
    96→    96→    enqueueJob(jobId)
    97→    97→    sendJson(res, 202, { ok: true, jobId, queued: true })
    98→    98→    return
    99→    99→  }
   100→   100→
   101→   101→  if (req.method === 'POST' && url === '/internal/cancel') {
   102→   102→    const { jobId } = body as { jobId?: string }
   103→   103→    if (!jobId) {
   104→   104→      sendJson(res, 400, { error: 'jobId required' })
   105→   105→      return
   106→   106→    }
   107→   107→    await cancelJob(jobId)
   108→   108→    sendJson(res, 200, { ok: true, jobId, cancelled: true })
   109→   109→    return
   110→   110→  }
   111→   111→
   112→   112→  sendJson(res, 404, { error: 'not found' })
   113→   113→}
   114→   114→
   115→   115→const httpServer = createServer((req, res) => {
   116→   116→  // This listener only runs if socket.io's engine.io middleware doesn't
   117→   117→  // intercept first. In practice, engine.io claims everything (path "/"),
   118→   118→  // so this fallback is only hit if engine.io is bypassed. We keep it as a
   119→   119→  // safety net.
   120→   120→  void httpHandler(req, res)
   121→   121→})
   122→   122→
   123→   123→const io = new Server(httpServer, {
   124→   124→  path: '/',
   125→   125→  cors: { origin: '*', methods: ['GET', 'POST', 'OPTIONS'] },
   126→   126→  pingTimeout: 60000,
   127→   127→  pingInterval: 25000,
   128→   128→})
   129→   129→
   130→   130→// Intercept /internal/* requests BEFORE engine.io processes them, since
   131→   131→// `path: '/'` makes engine.io claim every URL.
   132→   132→io.engine.use((req: any, res: any, next: any) => {
   133→   133→  const reqUrl: string = req.url || '/'
   134→   134→  const urlPath = reqUrl.split('?')[0]
   135→   135→  if (urlPath.startsWith('/internal/')) {
   136→   136→    void httpHandler(req, res)
   137→   137→    // Do NOT call next() — we handled it.
   138→   138→    return
   139→   139→  }
   140→   140→  next()
   141→   141→})

   144→   144→// Socket.io connection handling
   145→   145→// ---------------------------------------------------------------------------
   146→   146→
   147→   147→io.on('connection', (socket: Socket) => {
   148→   148→  console.log(`[io] connected ${socket.id}`)
   149→   149→
   150→   150→  socket.on('subscribe', async (payload: unknown) => {
   151→   151→    const jobId = extractJobId(payload)
   152→   152→    if (!jobId) return
   153→   153→    const room = `job:${jobId}`
   154→   154→    await socket.join(room)
   155→   155→    socket.emit('subscribed', { type: 'subscribed', jobId })
   156→   156→    // Immediately emit current status + recent logs.
   157→   157→    await emitStatus(jobId)
   158→   158→    await emitRecentLogs(jobId, socket)
   159→   159→  })
   160→   160→
   161→   161→  socket.on('unsubscribe', async (payload: unknown) => {
   162→   162→    const jobId = extractJobId(payload)
   163→   163→    if (!jobId) return
   164→   164→    await socket.leave(`job:${jobId}`)
   165→   165→  })
   166→   166→
   167→   167→  socket.on('cancel', async (payload: unknown) => {
   168→   168→    const jobId = extractJobId(payload)
   169→   169→    if (!jobId) return
   170→   170→    await cancelJob(jobId)
   171→   171→  })
   172→   172→
   173→   173→  socket.on('disconnect', () => {
   174→   174→    // rooms are auto-cleaned
   175→   175→  })
   176→   176→})

   178→   178→function extractJobId(payload: unknown): string | null {
   179→   179→  if (!payload) return null
   180→   180→  if (typeof payload === 'string') return payload
   181→   181→  if (typeof payload === 'object') {
   182→   182→    const p = payload as { jobId?: string }
   183→   183→    if (typeof p.jobId === 'string') return p.jobId
   184→   184→  }
   185→   185→  return null
   186→   186→}

   367→   367→// ---------------------------------------------------------------------------
   368→   368→// Job queue (one at a time)
   369→   369→// ---------------------------------------------------------------------------
   370→   370→
   371→   371→const queue: string[] = []
   372→   372→let currentlyRunning: string | null = null
   373→   373→const childProcesses = new Map<string, ChildProcess>()
   374→   374→const cancelledJobs = new Set<string>()
   375→   375→
   376→   376→function queueState() {
   377→   377→  return {
   378→   378→    queueLength: queue.length,
   379→   379→    currentlyRunning,
   380→   380→    cancelled: Array.from(cancelledJobs),
   381→   381→  }
   382→   382→}
   383→   383→
   384→   384→function enqueueJob(jobId: string) {
   385→   385→  if (queue.includes(jobId) || currentlyRunning === jobId) return
   386→   386→  queue.push(jobId)
   387→   387→  void processQueue()
   388→   388→}
   389→   389→
   390→   390→async function processQueue() {
   391→   391→  if (currentlyRunning) return
   392→   392→  const next = queue.shift()
   393→   393→  if (!next) return
   394→   394→  currentlyRunning = next
   395→   395→  try {
   396→   396→    await processJob(next)
   397→   397→  } catch (err) {
   398→   398→    console.error(`[queue] processJob threw for ${next}:`, err)
   399→   399→    try {
   400→   400→      await db.job.update({
   401→   401→        where: { id: next },
   402→   402→        data: {
   403→   403→          status: 'error',
   404→   404→          error: err instanceof Error ? err.message : String(err),
   405→   405→        },
   406→   406→      })
   407→   407→      io.to(`job:${next}`).emit('error', {
   408→   408→        type: 'error',
   409→   409→        jobId: next,
   410→   410→        error: err instanceof Error ? err.message : String(err),
   411→   411→      })
   412→   412→    } catch {
   413→   413→      // ignore
   414→   414→    }
   415→   415→  } finally {
   416→   416→    currentlyRunning = null
   417→   417→    cancelledJobs.delete(next)
   418→   418→    // Process the next queued job (if any).
   419→   419→    void processQueue()
   420→   420→  }
   421→   421→}

   427→   427→async function cancelJob(jobId: string): Promise<void> {
   428→   428→  cancelledJobs.add(jobId)
   429→   429→  const child = childProcesses.get(jobId)
   430→   430→  if (child) {
   431→   431→    try {
   432→   432→      child.kill('SIGTERM')
   433→   433→      // Force-kill after 3s if still alive.
   434→   434→      setTimeout(() => {
   435→   435→        try {
   436→   436→          child.kill('SIGKILL')
   437→   437→        } catch {
   438→   438→          // ignore
   439→   439→        }
   440→   440→      }, 3000)
   441→   441→    } catch {
   442→   442→      // ignore
   443→   443→    }
   444→   444→  }
   445→   445→  await db.job.update({
   446→   446→    where: { id: jobId },
   447→   447→    data: { status: 'cancelled', stage: 'cancelled', message: 'Cancelled by user' },
   448→   448→  }).catch(() => undefined)
   449→   449→  await emitLog(jobId, 'warn', 'cancel', 'Job cancelled by user.')
   450→   450→  io.to(`job:${jobId}`).emit('cancelled', { type: 'cancelled', jobId })
   451→   451→  await emitStatus(jobId)
   452→   452→}

   454→   454→// ---------------------------------------------------------------------------
   455→   455→// The core job pipeline
   456→   456→// ---------------------------------------------------------------------------
   457→   457→
   458→   458→async function processJob(jobId: string): Promise<void> {
   459→   459→  console.log(`[job:${jobId}] starting`)
   460→   460→  cancelledJobs.delete(jobId)
   461→   461→
   462→   462→  const job = await db.job.findUnique({
   463→   463→    where: { id: jobId },
   464→   464→    include: { chapters: { orderBy: { index: 'asc' } } },
   465→   465→  })
   466→   466→  if (!job) {
   467→   467→    console.error(`[job:${jobId}] not found`)
   468→   468→    return
   469→   469→  }
   470→   470→  if (job.status === 'cancelled') {
   471→   471→    console.log(`[job:${jobId}] already cancelled, skipping`)
   472→   472→    return
   473→   473→  }
   474→   474→
   475→   475→  // Prepare filesystem.
   476→   476→  await ensureDir(jobDir(jobId))
   477→   477→  await ensureDir(datasetDir(jobId))
   478→   478→  await ensureDir(workDir(jobId))
   479→   479→  await ensureDir(outputDir(jobId))
   480→   480→
   481→   481→  // -----------------------------
   482→   482→  // Phase 1: SCRAPE
   483→   483→  // -----------------------------
   484→   484→  await db.job.update({
   485→   485→    where: { id: jobId },
   486→   486→    data: { status: 'scraping', stage: 'scrape', message: 'Starting scrape' },
   487→   487→  })
   488→   488→  await emitStatus(jobId)
   489→   489→  await emitLog(jobId, 'info', 'scrape', `Starting scrape for "${job.mangaTitle}"`)
   490→   490→
   491→   491→  // If the API route did not pre-create Chapter rows, fetch them now.
   492→   492→  let chapters = job.chapters
   493→   493→  if (!chapters || chapters.length === 0) {
   494→   494→    const source = getSourceFromId(job.mangaId)
   495→   495→    if (!source) {
   496→   496→      throw new Error(`Cannot determine scraping source from manga ID: ${job.mangaId}`)
   497→   497→    }
   498→   498→    await emitLog(jobId, 'info', 'scrape', `Fetching chapter list from ${source}`)
   499→   499→    const fetched = await fetchChaptersForSource(source, job.mangaId, job.chapterLimit)
   500→   500→    if (fetched.length === 0) {
   501→   501→      throw new Error(`No chapters found for manga ${job.mangaId} on ${source}`)
   502→   502→    }
   503→   503→    // Create Chapter rows.
   504→   504→    for (let i = 0; i < fetched.length; i++) {
   505→   505→      const c = fetched[i]
   506→   506→      await db.chapter.create({
   507→   507→        data: {
   508→   508→          jobId,
   509→   509→          index: i + 1,
   510→   510→          mangadexId: c.mangadexId,
   511→   511→          chapterNum: c.chapterNum,
   512→   512→          title: c.title,
   513→   513→          language: c.language,

   653→   653→    'success',
   654→   654→    'scrape',

   711→   711→      const updated = await db.chapter.update({

   963→   963→      return
   964→   964→    }
   965→   965→    const outName = path.basename(outFile)
   966→   966→
   967→   967→    // -----------------------------
   968→   968→    // Reclaim disk space: intermediate work/ (sliced frames, per-panel
   969→   969→    // audio, per-chapter renders) and dataset/ (raw scraped chapter images)
   970→   970→    // are never needed again once the final video exists. For a large job
   971→   971→    // (e.g. 200 chapters) these dwarf the final compressed video, so this
   972→   972→    // alone is the biggest single win for disk usage.
   973→   973→    // -----------------------------
   974→   974→    for (const dir of [workDir(jobId), datasetDir(jobId)]) {
   975→   975→      try {
   976→   976→        await fs.rm(dir, { recursive: true, force: true })
   977→   977→      } catch (e) {
   978→   978→        await emitLog(jobId, 'warn', 'done', `Could not clean up ${dir}: ${e instanceof Error ? e.message : e}`)
   979→   979→      }
   980→   980→    }
   981→   981→    await emitLog(jobId, 'info', 'done', 'Cleaned up intermediate work/dataset files')
   982→   982→
   983→   983→    // -----------------------------
   984→   984→    // Offload the final video to Cloudflare R2, then free the local copy
   985→   985→    // too, so completed jobs don't keep accumulating local disk usage.
   986→   986→    // Only deletes the local file after the upload is verified to have
   987→   987→    // landed — if R2 isn't configured, or the upload fails, the local
   988→   988→    // file is left in place so the job is never left without a copy.
   989→   989→    // -----------------------------
   990→   990→    let r2Key: string | null = null
   991→   991→    if (isR2Configured()) {
   992→   992→      const key = `jobs/${jobId}/${outName}`
   993→   993→      try {
   994→   994→        await uploadFileToR2(outFile, key)
   995→   995→        await fs.rm(outFile, { force: true })
   996→   996→        r2Key = key
   997→   997→        await emitLog(jobId, 'success', 'done', `Uploaded output to R2 (${key}) and freed local copy`)
   998→   998→      } catch (e) {
   999→   999→        await emitLog(
  1000→  1000→          jobId,
  1001→  1001→          'warn',
  1002→  1002→          'done',
  1003→  1003→          `R2 upload failed, keeping local copy: ${e instanceof Error ? e.message : e}`,
  1004→  1004→        )
  1005→  1005→      }
  1006→  1006→    }
  1007→  1007→
  1008→  1008→    await db.job.update({
  1009→  1009→      where: { id: jobId },
  1010→  1010→      data: {
  1011→  1011→        status: 'done',
  1012→  1012→        progress: 100,
  1013→  1013→        stage: 'done',
  1014→  1014→        message: 'Pipeline complete',
  1015→  1015→        outputDir: outputDir(jobId),
  1016→  1016→        outputVideo: outName,
  1017→  1017→        r2Key,
  1018→  1018→      },
  1019→  1019→    })
  1020→  1020→    await emitLog(jobId, 'success', 'done', `Pipeline complete. Output: ${outName}`)
  1021→  1021→    io.to(`job:${jobId}`).emit('done', { type: 'done', jobId, outputVideo: outName })
  1022→  1022→    await emitStatus(jobId)
  1023→  1023→    console.log(`[job:${jobId}] done`)
  1024→  1024→  } else {
  1025→  1025→    const err = `master_pipeline.py exited with code ${exitCode}`
  1026→  1026→    await db.job.update({
  1027→  1027→      where: { id: jobId },
  1028→  1028→      data: { status: 'error', stage: 'render', error: err, message: err },
  1029→  1029→    })
  1030→  1030→    await emitLog(jobId, 'error', 'render', err)
  1031→  1031→    io.to(`job:${jobId}`).emit('error', { type: 'error', jobId, error: err })
  1032→  1032→    await emitStatus(jobId)
  1033→  1033→    console.error(`[job:${jobId}] failed: ${err}`)
  1034→  1034→  }
  1035→  1035→}

  1041→  1041→async function readBody(req: IncomingMessage): Promise<any> {
  1042→  1042→  return await new Promise((resolve) => {
  1043→  1043→    const chunks: Buffer[] = []
  1044→  1044→    req.on('data', (c) => chunks.push(c as Buffer))
  1045→  1045→    req.on('end', () => {
  1046→  1046→      const raw = Buffer.concat(chunks).toString('utf8')
  1047→  1047→      if (!raw) {
  1048→  1048→        resolve({})
  1049→  1049→        return
  1050→  1050→      }
  1051→  1051→      try {
  1052→  1052→        resolve(JSON.parse(raw))
  1053→  1053→      } catch {
  1054→  1054→        resolve({})
  1055→  1055→      }
  1056→  1056→    })
  1057→  1057→    req.on('error', () => resolve({}))
  1058→  1058→  })
  1059→  1059→}

  1078→  1078→async function shutdown(signal: string) {
  1079→  1079→  console.log(`[pipeline-service] received ${signal}, shutting down`)
  1080→  1080→  // Kill any running subprocesses.
  1081→  1081→  for (const [jobId, child] of childProcesses.entries()) {
  1082→  1082→    try {
  1083→  1083→      child.kill('SIGTERM')
  1084→  1084→      console.log(`[pipeline-service] killed subprocess for job ${jobId}`)
  1085→  1085→    } catch {
  1086→  1086→      // ignore
  1087→  1087→    }
  1088→  1088→  }
  1089→  1089→  childProcesses.clear()
  1090→  1090→  // Close socket.io + HTTP server.
  1091→  1091→  io.close()

  1093→  1093→    console.log('[pipeline-service] http server closed')
  1094→  1094→    process.exit(0)
  1095→  1095→  })
  1096→  1096→  // Force-exit after 5s if close hangs.
  1097→  1097→  setTimeout(() => process.exit(0), 5000).unref()
  1098→  1098→}
