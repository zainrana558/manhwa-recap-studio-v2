    12→    12→

    14→    14→

   185→   185→def natural_sort_key(p: Path):
   186→   186→    return [int(t) if t.isdigit() else t.lower() for t in re.split(r"(\d+)", p.stem)]

   188→   188→
   189→   189→def discover_chapters(cfg: PipelineConfig) -> List[Chapter]:
   190→   190→    """Scan input_dir for chapter_XXX folders, sort numerically, load panels + summary."""
   191→   191→    if not cfg.input_dir.exists():
   192→   192→        raise FileNotFoundError(f"Input directory not found: {cfg.input_dir}")
   193→   193→
   194→   194→    chapter_dirs = sorted(
   195→   195→        [d for d in cfg.input_dir.iterdir() if d.is_dir()],
   196→   196→        key=natural_sort_key,
   197→   197→    )
   198→   198→    if not chapter_dirs:
   199→   199→        raise FileNotFoundError(f"No chapter subfolders found in {cfg.input_dir}")
   200→   200→
   201→   201→    if cfg.total_chapters:
   202→   202→        chapter_dirs = chapter_dirs[: cfg.total_chapters]
   203→   203→
   204→   204→    chapters: List[Chapter] = []
   205→   205→    for i, d in enumerate(chapter_dirs, start=1):
   206→   206→        panels = sorted(
   207→   207→            [p for p in d.iterdir() if p.suffix.lower() in IMAGE_EXTS],
   208→   208→            key=natural_sort_key,
   209→   209→        )
   210→   210→        if not panels:
   211→   211→            log.warning("Chapter folder %s has no images — skipping", d.name)
   212→   212→            continue
   213→   213→
   214→   214→        # Load per-image narrations if available (preferred), else fall back to summary.txt.
   215→   215→        narration_file = d / "narration.json"
   216→   216→        summary_text = ""
   217→   217→        image_narrations: dict = {}
   218→   218→        if narration_file.exists():
   219→   219→            try:
   220→   220→                narr_data = json.loads(narration_file.read_text(encoding="utf-8"))
   221→   221→                if isinstance(narr_data, list):
   222→   222→                    for item in narr_data:
   223→   223→                        if isinstance(item, dict) and "image" in item and "text" in item:
   224→   224→                            image_narrations[item["image"]] = item["text"]
   225→   225→                    log.info("[%s] loaded %d per-image narrations from narration.json", d.name, len(image_narrations))
   226→   226→            except Exception as e:
   227→   227→                log.warning("[%s] failed to parse narration.json: %s", d.name, e)
   228→   228→
   229→   229→        if not image_narrations:
   230→   230→            summary_file = d / "summary.txt"
   231→   231→            summary_text = summary_file.read_text(encoding="utf-8").strip() if summary_file.exists() else ""
   232→   232→            if not summary_text:
   233→   233→                log.warning(
   234→   234→                    "No narration.json or summary.txt in %s — narrative rewrite will run on empty text",
   235→   235→                    d.name,
   236→   236→                )
   237→   237→
   238→   238→        chapters.append(
   239→   239→            Chapter(
   240→   240→                index=i, name=d.name, folder=d, panel_paths=panels,
   241→   241→                summary_text=summary_text, image_narrations=image_narrations,
   242→   242→            )
   243→   243→        )
   244→   244→
   245→   245→    log.info("Discovered %d chapters (%d total panel images)", len(chapters), sum(len(c.panel_paths) for c in chapters))
   246→   246→    return chapters

   250→   250→# 2a. FAST FRAME PREPARATION (no slicing — one frame per source image)
   251→   251→# ---------------------------------------------------------------------------
   252→   252→
   253→   253→def _fast_prepare_frames(cfg: PipelineConfig, chapter: Chapter) -> List[tuple]:
   254→   254→    """Prepare one 1920x1080 canvas frame per source panel image, centered and
   255→   255→    scaled to fit. This avoids the expensive slicing step for tall webtoon
   256→   256→    strips. Each image gets exactly one frame that is displayed for the full
   257→   257→    duration of its narration TTS, keeping voice and image perfectly in sync.
   258→   258→
   259→   259→    Returns list of (frame_path, source_panel_index) tuples.
   260→   260→    Resumable: reuses existing frames from the slices directory."""
   261→   261→    out_dir = cfg.temp_slices_dir / chapter.tag
   262→   262→    manifest_path = out_dir / "manifest.json"
   263→   263→    if manifest_path.exists():
   264→   264→        try:
   265→   265→            data = json.loads(manifest_path.read_text())
   266→   266→            frames = data["frames"]
   267→   267→            sources = data.get("sources")
   268→   268→            if sources and len(sources) == len(frames):
   269→   269→                log.info("[%s] frames already exist (%d frames) — skipping", chapter.tag, len(frames))
   270→   270→                return [(Path(f), s) for f, s in zip(frames, sources)]
   271→   271→        except Exception:
   272→   272→            pass
   273→   273→
   274→   274→    out_dir.mkdir(parents=True, exist_ok=True)
   275→   275→    frame_data: List[tuple] = []
   276→   276→
   277→   277→    for panel_idx, panel_path in enumerate(chapter.panel_paths):
   278→   278→        fp = out_dir / f"frame_{panel_idx:05d}.jpg"
   279→   279→        if fp.exists():
   280→   280→            frame_data.append((fp, panel_idx))
   281→   281→            continue
   282→   282→        frame = _compose_canvas_from_source(panel_path)
   283→   283→        frame.save(fp, quality=92)
   284→   284→        frame_data.append((fp, panel_idx))
   285→   285→
   286→   286→    manifest_path.write_text(json.dumps({
   287→   287→        "frames": [str(f) for f, _ in frame_data],
   288→   288→        "sources": [s for _, s in frame_data],
   289→   289→    }, indent=2))
   290→   290→    log.info("[%s] prepared %d frames (one per source panel)", chapter.tag, len(frame_data))
   291→   291→    return frame_data

   294→   294→def _compose_canvas_from_source(panel_path: Path):
   295→   295→    """Load a source panel image and center it on a 1920x1080 canvas.
   296→   296→    For tall webtoon strips, crops the center section directly from the
   297→   297→    original resolution (avoids expensive full-width resize)."""
   298→   298→    from PIL import Image
   299→   299→
   300→   300→    img = Image.open(panel_path).convert("RGB")
   301→   301→    cw, ch = img.size
   302→   302→
   303→   303→    if ch / cw > 1.5:
   304→   304→        # Tall strip: crop center section from ORIGINAL resolution,
   305→   305→        # then resize only the crop to 1920x1080. This avoids
   306→   306→        # creating a 1920x33000 intermediate image.
   307→   307→        crop_h = int(cw * CANVAS_H / CANVAS_W)
   308→   308→        crop_top = max(0, (ch - crop_h) // 2)
   309→   309→        crop_bot = min(ch, crop_top + crop_h)
   310→   310→        crop = img.crop((0, crop_top, cw, crop_bot))
   311→   311→        fg = crop.resize((CANVAS_W, CANVAS_H), Image.LANCZOS)
   312→   312→    else:
   313→   313→        # Normal aspect ratio: fit within canvas.
   314→   314→        fg_scale = min(CANVAS_W / cw, CANVAS_H / ch)
   315→   315→        fw, fh = max(1, int(cw * fg_scale)), max(1, int(ch * fg_scale))
   316→   316→        fg = img.resize((fw, fh), Image.LANCZOS)
   317→   317→
   318→   318→    # Dark background.
   319→   319→    canvas = Image.new("RGB", (CANVAS_W, CANVAS_H), (20, 20, 30))
   320→   320→
   321→   321→    # Center the image on canvas.
   322→   322→    fx = (CANVAS_W - fg.width) // 2
   323→   323→    fy = (CANVAS_H - fg.height) // 2
   324→   324→    canvas.paste(fg, (fx, fy))
   325→   325→
   326→   326→    return canvas

   341→   341→def _detect_panel_gutters(img_gray, y_offset: int = 0) -> List[int]:
   342→   342→    """Detect horizontal gutter lines (panel separators) in a webtoon strip.
   343→   343→
   344→   344→    Uses a multi-signal approach:
   345→   345→      1. Row variance (flat/uniform rows = gutters)
   346→   346→      2. Row brightness (white gaps have high mean, black borders have low mean)
   347→   347→
   348→   348→    A row is classified as gutter if it satisfies EITHER:
   349→   349→      - Low variance (uniform color) AND (very bright OR very dark)
   350→   350→      - Very low variance (nearly uniform regardless of color)
   351→   351→
   352→   352→    Returns sorted list of ABSOLUTE y-coordinates for cuts.
   353→   353→    """
   354→   354→    import cv2
   355→   355→    import numpy as np
   356→   356→
   357→   357→    h, w = img_gray.shape
   358→   358→    if h < 10:
   359→   359→        return []
   360→   360→
   361→   361→    row_means = np.mean(img_gray, axis=1).astype(np.float32)
   362→   362→    row_stds = np.std(img_gray, axis=1).astype(np.float32)
   363→   363→
   364→   364→    # Smooth both signals to avoid single-row noise.
   365→   365→    kernel = max(3, min(7, h // 300))
   366→   366→    row_means = cv2.blur(row_means.reshape(-1, 1), (kernel, 1)).flatten()
   367→   367→    row_stds = cv2.blur(row_stds.reshape(-1, 1), (kernel, 1)).flatten()
   368→   368→
   369→   369→    # Classify each row as gutter or not.
   370→   370→    # Strategy: a gutter row is typically uniform (low std) AND is either
   371→   371→    # very bright (white gap >200) or very dark (black border <60) or
   372→   372→    # extremely uniform (std < 3 regardless of color).
   373→   373→    is_bright_gutter = (row_stds < 12) & (row_means > 200)
   374→   374→    is_dark_gutter = (row_stds < 12) & (row_means < 60)
   375→   375→    is_flat_gutter = row_stds < 3
   376→   376→    is_gutter = is_bright_gutter | is_dark_gutter | is_flat_gutter
   377→   377→
   378→   378→    # Extract gutter runs (consecutive gutter rows).
   379→   379→    cuts: List[int] = []
   380→   380→    run_start = None
   381→   381→    # Minimum gutter height: at least 2px, or proportional to image height.
   382→   382→    min_gutter_height = max(2, h // 500)
   383→   383→    # Minimum distance between cuts to avoid too many cuts close together.
   384→   384→    min_panel_height = max(30, h // 100)
   385→   385→
   386→   386→    for i, g in enumerate(is_gutter):
   387→   387→        if g and run_start is None:
   388→   388→            run_start = i
   389→   389→        elif not g and run_start is not None:
   390→   390→            if i - run_start >= min_gutter_height:
   391→   391→                cut_pos = y_offset + (run_start + i) // 2
   392→   392→                # Ensure minimum panel height between cuts.
   393→   393→                if not cuts or (cut_pos - cuts[-1]) >= min_panel_height:
   394→   394→                    cuts.append(cut_pos)
   395→   395→            run_start = None
   396→   396→    if run_start is not None and h - run_start >= min_gutter_height:
   397→   397→        cut_pos = y_offset + (run_start + h) // 2
   398→   398→        if not cuts or (cut_pos - cuts[-1]) >= min_panel_height:
   399→   399→            cuts.append(cut_pos)
   400→   400→
   401→   401→    return cuts
   402→   402→

   404→   404→def _detect_vertical_gutters(img_gray) -> List[int]:
   405→   405→    """Detect vertical gutter lines (column separators) in a manga row.
   406→   406→
   407→   407→    Scans columns for uniform vertical bands — blank white gaps or thin
   408→   408→    black borders between side-by-side panels. Returns sorted list of
   409→   409→    absolute x-coordinates where the row should be cut vertically.
   410→   410→    """
   411→   411→    import cv2
   412→   412→    import numpy as np
   413→   413→
   414→   414→    h, w = img_gray.shape
   415→   415→    if w < 10:
   416→   416→        return []
   417→   417→
   418→   418→    col_stds = np.std(img_gray, axis=0).astype(np.float32)
   419→   419→    # Smooth to avoid single-column noise spikes.
   420→   420→    kernel = max(1, min(5, w // 500))
   421→   421→    if kernel > 1:
   422→   422→        col_stds = cv2.blur(col_stds.reshape(1, -1), (1, kernel)).flatten()
   423→   423→
   424→   424→    # Adaptive threshold for vertical gutters.
   425→   425→    gutter_threshold = float(min(14.0, max(6.0, np.percentile(col_stds, 20))))
   426→   426→    is_gutter = col_stds < gutter_threshold
   427→   427→
   428→   428→    cuts: List[int] = []
   429→   429→    run_start = None
   430→   430→    min_gutter_width = max(2, w // 800)
   431→   431→    for i, g in enumerate(is_gutter):
   432→   432→        if g and run_start is None:
   433→   433→            run_start = i
   434→   434→        elif not g and run_start is not None:
   435→   435→            if i - run_start >= min_gutter_width:
   436→   436→                cuts.append((run_start + i) // 2)
   437→   437→            run_start = None
   438→   438→    if run_start is not None and w - run_start >= min_gutter_width:
   439→   439→        cuts.append((run_start + w) // 2)
   440→   440→    return cuts

   443→   443→def _split_into_panel_segments(img_gray, max_segment_height: int) -> List[tuple]:
   444→   444→    """Return a list of (top, bottom) panel boundaries for a webtoon strip.
   445→   445→
   446→   446→    Uses multi-pass gutter detection:
   447→   447→      1. Primary pass with strict thresholds
   448→   448→      2. Re-scan oversized segments (>1.5x max height) with relaxed thresholds
   449→   449→      3. Re-scan still-oversized segments with very relaxed thresholds
   450→   450→
   451→   451→    Each segment represents one manga panel.
   452→   452→    """
   453→   453→    import numpy as np
   454→   454→
   455→   455→    h, w = img_gray.shape
   456→   456→    gutter_cuts = _detect_panel_gutters(img_gray)
   457→   457→    boundaries = [0] + gutter_cuts + [h]
   458→   458→    segments = []
   459→   459→    for i in range(len(boundaries) - 1):
   460→   460→        top, bot = boundaries[i], boundaries[i + 1]
   461→   461→        if bot - top < 50:
   462→   462→            continue
   463→   463→        segments.append((top, bot))
   464→   464→    if not segments:
   465→   465→        segments = [(0, h)]
   466→   466→
   467→   467→    # Recursive re-scanning of oversized segments.
   468→   468→    refined = _refine_segments(img_gray, segments, max_segment_height, depth=0)
   469→   469→    return refined

   471→   471→
   472→   472→def _refine_segments(img_gray, segments: List[tuple], max_height: int, depth: int) -> List[tuple]:
   473→   473→    """Re-scan oversized segments with progressively relaxed thresholds."""
   474→   474→    import numpy as np
   475→   475→
   476→   476→    if depth >= 2:  # Max 2 re-scan passes
   477→   477→        return segments
   478→   478→
   479→   479→    refined: List[tuple] = []
   480→   480→    for top, bot in segments:
   481→   481→        seg_h = bot - top
   482→   482→        if seg_h <= max_height * 1.5:
   483→   483→            refined.append((top, bot))
   484→   484→            continue
   485→   485→
   486→   486→        # Re-scan this sub-region with relaxed thresholds.
   487→   487→        sub = img_gray[top:bot, :]
   488→   488→        sub_stds = np.std(sub, axis=1).astype(np.float32)
   489→   489→        sub_means = np.mean(sub, axis=1).astype(np.float32)
   490→   490→
   491→   491→        kernel = max(3, min(7, seg_h // 300))
   492→   492→        sub_stds = cv2.blur(sub_stds.reshape(-1, 1), (kernel, 1)).flatten()
   493→   493→        sub_means = cv2.blur(sub_means.reshape(-1, 1), (kernel, 1)).flatten()
   494→   494→
   495→   495→        # Progressively relax the thresholds with each pass.
   496→   496→        if depth == 0:
   497→   497→            std_thresh = float(min(18.0, max(8.0, np.percentile(sub_stds, 25))))
   498→   498→        else:
   499→   499→            std_thresh = float(min(25.0, max(12.0, np.percentile(sub_stds, 35))))
   500→   500→
   501→   501→        is_gutter = sub_stds < std_thresh
   502→   502→        sub_cuts: List[int] = []
   503→   503→        run_start = None
   504→   504→        min_h = max(2, seg_h // 600)
   505→   505→        for i, g in enumerate(is_gutter):
   506→   506→            if g and run_start is None:
   507→   507→                run_start = i
   508→   508→            elif not g and run_start is not None:
   509→   509→                if i - run_start >= min_h:
   510→   510→                    sub_cuts.append((run_start + i) // 2)
   511→   511→                run_start = None
   512→   512→        if run_start is not None and seg_h - run_start >= min_h:
   513→   513→            sub_cuts.append((run_start + seg_h) // 2)
   514→   514→
   515→   515→        if sub_cuts:
   516→   516→            sub_boundaries = [0] + sub_cuts + [seg_h]
   517→   517→            new_segs = []
   518→   518→            for j in range(len(sub_boundaries) - 1):
   519→   519→                st, sb = sub_boundaries[j], sub_boundaries[j + 1]
   520→   520→                if sb - st >= 50:
   521→   521→                    new_segs.append((top + st, top + sb))
   522→   522→            if new_segs:
   523→   523→                refined.extend(_refine_segments(img_gray, new_segs, max_height, depth + 1))
   524→   524→                continue
   525→   525→        refined.append((top, bot))
   526→   526→
   527→   527→    return refined

   530→   530→def _inpaint_bubbles(img) -> "object":
   531→   531→    """Detect and inpaint speech bubbles/text in a panel image.
   532→   532→
   533→   533→    Uses OpenCV to find text regions (light blobs with borders), then fills
   534→   534→    them with surrounding pixels via OpenCV's inpainting (Telea algorithm —
   535→   535→    a free, fast content-aware fill that reconstructs background art).
   536→   536→    """
   537→   537→    import cv2
   538→   538→    import numpy as np
   539→   539→    from PIL import Image
   540→   540→
   541→   541→    arr = np.array(img)  # RGB
   542→   542→    gray = cv2.cvtColor(arr, cv2.COLOR_RGB2GRAY)
   543→   543→
   544→   544→    # Detect text/bubbles: white-ish blobs (speech bubbles are usually white).
   545→   545→    # Threshold to find bright regions.
   546→   546→    _, bright = cv2.threshold(gray, 240, 255, cv2.THRESH_BINARY)
   547→   547→    # Morphological close to merge nearby text into blob masks.
   548→   548→    kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (15, 15))
   549→   549→    bright = cv2.morphologyEx(bright, cv2.MORPH_CLOSE, kernel)
   550→   550→    # Find contours that look like text/bubbles (area in range).
   551→   551→    contours, _ = cv2.findContours(bright, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
   552→   552→    mask = np.zeros(gray.shape, dtype=np.uint8)
   553→   553→    for c in contours:
   554→   554→        area = cv2.contourArea(c)
   555→   555→        if INPAINT_MIN_BUBBLE_AREA < area < INPAINT_MAX_BUBBLE_AREA:
   556→   556→            cv2.drawContours(mask, [c], -1, 255, -1)
   557→   557→            # Dilate the mask slightly so inpainting covers edges cleanly.
   558→   558→    if mask.any():
   559→   559→        mask = cv2.dilate(mask, kernel, iterations=2)
   560→   560→        # Inpaint using the Telea algorithm (fast, free, content-aware).
   561→   561→        arr = cv2.inpaint(arr, mask, 5, cv2.INPAINT_TELEA)
   562→   562→    return Image.fromarray(arr)

   565→   565→def slice_chapter_panels(cfg: PipelineConfig, chapter: Chapter) -> List[tuple]:
   566→   566→    """
   567→   567→    Slice each source image into individual panel frames.
   568→   568→
   569→   569→    Strategy:
   570→   570→      1. Detect gutter boundaries on ORIGINAL resolution
   571→   571→      2. For each detected panel:
   572→   572→         a. If aspect ratio is reasonable (≤2.5:1), show as ONE frame (contain-fit)
   573→   573→         b. If aspect ratio is very tall (>2.5:1), split into equal-height sub-frames
   574→   574→            so each sub-frame has a readable size on screen (≥50% of canvas width)
   575→   575→      3. All frames use contain-fit — nothing ever goes out of frame
   576→   576→      4. For landscape segments, detect vertical gutters for side-by-side panels
   577→   577→
   578→   578→    Returns list of (frame_path, source_panel_index) tuples in order.
   579→   579→    Resumable: if the chapter's slice folder has a manifest, reuse it.
   580→   580→    """
   581→   581→    from PIL import Image, ImageFilter
   582→   582→
   583→   583→    out_dir = cfg.temp_slices_dir / chapter.tag
   584→   584→    manifest_path = out_dir / "manifest.json"
   585→   585→    if manifest_path.exists():
   586→   586→        try:
   587→   587→            data = json.loads(manifest_path.read_text())
   588→   588→            frames = data["frames"]
   589→   589→            sources = data.get("sources")
   590→   590→            if sources and len(sources) == len(frames):
   591→   591→                log.info("[%s] slices already exist (%d frames) — skipping", chapter.tag, len(frames))
   592→   592→                return [(Path(f), s) for f, s in zip(frames, sources)]
   593→   593→            log.info("[%s] old manifest format — re-slicing", chapter.tag)
   594→   594→        except Exception:
   595→   595→            pass
   596→   596→
   597→   597→    out_dir.mkdir(parents=True, exist_ok=True)
   598→   598→    frame_data: List[tuple] = []
   599→   599→    frame_counter = 0
   600→   600→
   601→   601→    for panel_idx, panel_path in enumerate(chapter.panel_paths):
   602→   602→        with Image.open(panel_path) as img:
   603→   603→            img = img.convert("RGB")
   604→   604→            w, h = img.size
   605→   605→
   606→   606→            # Detect gutters on ORIGINAL resolution for accuracy.
   607→   607→            import cv2
   608→   608→            import numpy as np
   609→   609→            gray = cv2.cvtColor(np.array(img), cv2.COLOR_RGB2GRAY)
   610→   610→            segments = _split_into_panel_segments(gray, max_segment_height=h * 0.6)
   611→   611→            log.debug("[%s] img %d: split into %d panel segment(s)", chapter.tag, panel_idx, len(segments))
   612→   612→
   613→   613→            for seg_top, seg_bot in segments:
   614→   614→                seg_h = seg_bot - seg_top
   615→   615→                if seg_h < 30:
   616→   616→                    continue
   617→   617→
   618→   618→                # Check for vertical gutters in landscape segments.
   619→   619→                sub_regions: List[tuple] = []  # (left, top, right, bottom)
   620→   620→                aspect = seg_h / w if w > 0 else 1
   621→   621→                if aspect < 0.5:  # wide landscape segment
   622→   622→                    seg_gray = gray[seg_top:seg_bot, :]
   623→   623→                    v_cuts = _detect_vertical_gutters(seg_gray)
   624→   624→                    if len(v_cuts) >= 1:
   625→   625→                        v_boundaries = [0] + v_cuts + [w]
   626→   626→                        for vi in range(len(v_boundaries) - 1):
   627→   627→                            vl, vr = v_boundaries[vi], v_boundaries[vi + 1]
   628→   628→                            if vr - vl >= w * 0.1:
   629→   629→                                sub_regions.append((vl, seg_top, vr, seg_bot))
   630→   630→                    if not sub_regions:
   631→   631→                        sub_regions.append((0, seg_top, w, seg_bot))
   632→   632→                else:
   633→   633→                    sub_regions.append((0, seg_top, w, seg_bot))
   634→   634→
   635→   635→                for left, top, right, bottom in sub_regions:
   636→   636→                    crop = img.crop((left, top, right, bottom))
   637→   637→                    cw, ch = crop.size
   638→   638→                    crop_aspect = ch / cw if cw > 0 else 1
   639→   639→
   640→   640→                    if crop_aspect <= 2.5:
   641→   641→                        # Reasonable aspect ratio — one frame, contain-fit.
   642→   642→                        frame = _compose_canvas(crop, ImageFilter=ImageFilter)
   643→   643→                        fp = out_dir / f"frame_{frame_counter:05d}.jpg"
   644→   644→                        frame.save(fp, quality=92)
   645→   645→                        frame_data.append((fp, panel_idx))
   646→   646→                        frame_counter += 1
   647→   647→                    else:
   648→   648→                        # Very tall panel — split into sub-frames so each
   649→   649→                        # sub-frame is at a readable aspect ratio (≤2:1).
   650→   650→                        # This ensures each sub-frame fills ≥50% of canvas width.
   651→   651→                        n_splits = max(2, int(crop_aspect / 1.8))
   652→   652→                        sub_h = ch / n_splits
   653→   653→                        for si in range(n_splits):
   654→   654→                            sub_top = int(si * sub_h)
   655→   655→                            sub_bot = min(ch, int((si + 1) * sub_h))
   656→   656→                            sub_crop = crop.crop((0, sub_top, cw, sub_bot))
   657→   657→                            frame = _compose_canvas(sub_crop, ImageFilter=ImageFilter)
   658→   658→                            fp = out_dir / f"frame_{frame_counter:05d}.jpg"
   659→   659→                            frame.save(fp, quality=92)
   660→   660→                            frame_data.append((fp, panel_idx))
   661→   661→                            frame_counter += 1
   662→   662→
   663→   663→    manifest_path.write_text(json.dumps({
   664→   664→        "frames": [str(f) for f, _ in frame_data],
   665→   665→        "sources": [s for _, s in frame_data],
   666→   666→    }, indent=2))
   667→   667→    log.info("[%s] sliced %d source panels into %d frames (panel-by-panel, contain-fit)",
   668→   668→             chapter.tag, len(chapter.panel_paths), len(frame_data))
   669→   669→    return frame_data

   672→   672→def _compose_canvas(crop, ImageFilter):
   673→   673→    """Center `crop` on a fixed 1920x1080 canvas, filling empty space with a
   674→   674→    Gaussian-blurred, darkened version of the crop itself (no hard letterboxing)."""
   675→   675→    from PIL import Image, ImageEnhance
   676→   676→
   677→   677→    cw, ch = crop.size
   678→   678→
   679→   679→    # Background: cover-fit blur of the same crop, darkened.
   680→   680→    bg_scale = max(CANVAS_W / cw, CANVAS_H / ch)
   681→   681→    bg = crop.resize((max(1, int(cw * bg_scale)), max(1, int(ch * bg_scale))), Image.LANCZOS)
   682→   682→    bg = bg.filter(ImageFilter.GaussianBlur(15))
   683→   683→    bg = ImageEnhance.Brightness(bg).enhance(0.45)
   684→   684→    # Center-crop the blurred bg to canvas size.
   685→   685→    bx = (bg.width - CANVAS_W) // 2
   686→   686→    by = (bg.height - CANVAS_H) // 2
   687→   687→    bg = bg.crop((bx, by, bx + CANVAS_W, by + CANVAS_H))
   688→   688→
   689→   689→    canvas = Image.new("RGB", (CANVAS_W, CANVAS_H))
   690→   690→    canvas.paste(bg, (0, 0))
   691→   691→
   692→   692→    # Foreground: fit crop inside canvas preserving aspect ratio, centered.
   693→   693→    fg_scale = min(CANVAS_W / cw, CANVAS_H / ch)
   694→   694→    fw, fh = max(1, int(cw * fg_scale)), max(1, int(ch * fg_scale))
   695→   695→    fg = crop.resize((fw, fh), Image.LANCZOS)
   696→   696→    fx = (CANVAS_W - fw) // 2
   697→   697→    fy = (CANVAS_H - fh) // 2
   698→   698→    canvas.paste(fg, (fx, fy))
   699→   699→
   700→   700→    return canvas

   745→   745→        if not translated:

   749→   749→        return translated
   750→   750→    except Exception as e:
   751→   751→        log.warning("[%s] Groq translation failed (%s) — using raw text as-is", cache_tag, e)
   752→   752→        out_path.write_text(text, encoding="utf-8")

   901→   901→def split_into_segments(text: str, n: int) -> List[str]:
   902→   902→    """Split narration text into n roughly-equal word chunks, one per frame,
   903→   903→    so every sliced image gets its own narration segment for tight sync.
   904→   904→    If there are fewer words than frames, trailing frames get an empty
   905→   905→    segment and simply hold silently for SILENT_FRAME_DURATION."""
   906→   906→    words = text.split()
   907→   907→    if n <= 0:
   908→   908→        return []
   909→   909→    if not words:
   910→   910→        return [""] * n
   911→   911→
   912→   912→    segments: List[str] = []
   913→   913→    idx = 0
   914→   914→    total = len(words)
   915→   915→    for i in range(n):
   916→   916→        remaining_slots = n - i
   917→   917→        remaining_words = total - idx
   918→   918→        if remaining_words <= 0:
   919→   919→            segments.append("")
   920→   920→            continue
   921→   921→        seg_len = max(1, round(remaining_words / remaining_slots))
   922→   922→        segments.append(" ".join(words[idx: idx + seg_len]))
   923→   923→        idx += seg_len
   924→   924→    return segments

  1004→  1004→def build_chapter_audio_track(cfg: PipelineConfig, chapter: Chapter, segment_audio_paths: List[Path]) -> Path:
  1005→  1005→    """Concatenate per-segment (per-panel) audio clips into one continuous
  1006→  1006→    chapter audio track, then apply Phase 3 audio post-processing:
  1007→  1007→      1. Compressor (fast attack, 3:1 ratio) to even out spoken volume.
  1008→  1008→      2. EQ boost (+2dB at 80-120Hz) for deep narrative gravitas.
  1009→  1009→      3. Loudness normalization to -14 LUFS (YouTube standard).
  1010→  1010→    Because each input clip is now a full continuous utterance (not a lone
  1011→  1011→    word), the only joins left are the natural breath-like gaps between
  1012→  1012→    panels/scenes — not mid-sentence chops. Resumable."""
  1013→  1013→    out_path = cfg.temp_audio_dir / f"{chapter.tag}.mp3"
  1014→  1014→    if out_path.exists():
  1015→  1015→        log.info("[%s] chapter audio track already assembled — skipping", chapter.tag)
  1016→  1016→        return out_path
  1017→  1017→
  1018→  1018→    concat_list = cfg.temp_audio_dir / f"{chapter.tag}_audio_concat.txt"
  1019→  1019→    with concat_list.open("w", encoding="utf-8") as f:
  1020→  1020→        for p in segment_audio_paths:
  1021→  1021→            f.write(f"file '{p.resolve().as_posix()}'\n")
  1022→  1022→
  1023→  1023→    # First: raw concat (stream copy).
  1024→  1024→    raw_path = cfg.temp_audio_dir / f"{chapter.tag}_raw.mp3"
  1025→  1025→    run_ffmpeg(
  1026→  1026→        [
  1027→  1027→            "ffmpeg", "-y", "-f", "concat", "-safe", "0",
  1028→  1028→            "-i", str(concat_list),
  1029→  1029→            "-c", "copy",
  1030→  1030→            str(raw_path),
  1031→  1031→        ]
  1032→  1032→    )
  1033→  1033→
  1034→  1034→    # Second: audio post-processing — compressor + EQ + loudnorm.
  1035→  1035→    # acompressor: fast attack (5ms), 3:1 ratio, moderate threshold for even voice.
  1036→  1036→    # bass: +2dB low-shelf boost at 100Hz for narrative gravitas.
  1037→  1037→    # loudnorm: EBU R128 normalization to -14 LUFS (YouTube standard).
  1038→  1038→    #
  1039→  1039→    # NOTE: if a chapter has no narration at all (every segment fell back to
  1040→  1040→    # silence — narration.json entries all empty, or narrative rewrite
  1041→  1041→    # returned nothing), raw_path is pure digital silence end-to-end.
  1042→  1042→    # loudnorm's gain boost on an absolutely-zero signal trips a real
  1043→  1043→    # libmp3lame bug (psymodel.c calc_energy: Assertion 'el >= 0' failed,
  1044→  1044→    # verified locally), which crashes ffmpeg outright. There's nothing to
  1045→  1045→    # compress/normalize in silence anyway, so on that failure we just copy
  1046→  1046→    # the raw (silent) track through untouched instead of crashing the job.
  1047→  1047→    af = (
  1048→  1048→        "acompressor=attack=5:release=80:ratio=3:threshold=-20dB:makeup=2,"
  1049→  1049→        "bass=gain=2:frequency=100:width=80,"
  1050→  1050→        f"loudnorm=I={TARGET_LOUDNESS_LUFS}:TP=-1.5:LRA=11"
  1051→  1051→    )
  1052→  1052→    try:
  1053→  1053→        run_ffmpeg(
  1054→  1054→            [
  1055→  1055→                "ffmpeg", "-y", "-i", str(raw_path),
  1056→  1056→                "-af", af,
  1057→  1057→                "-ar", str(AUDIO_SAMPLE_RATE), "-b:a", AUDIO_BITRATE, "-ac", "1",
  1058→  1058→                str(out_path),
  1059→  1059→            ]
  1060→  1060→        )
  1061→  1061→    except RuntimeError as e:
  1062→  1062→        log.warning(
  1063→  1063→            "[%s] audio post-processing failed (%s) — falling back to raw (unprocessed) audio track",
  1064→  1064→            chapter.tag, e,
  1065→  1065→        )
  1066→  1066→        shutil.copy2(raw_path, out_path)
  1067→  1067→    raw_path.unlink(missing_ok=True)
  1068→  1068→    log.info("[%s] assembled + post-processed chapter audio (compressor+EQ+loudnorm) -> %s", chapter.tag, out_path.name)
  1069→  1069→    return out_path

  1108→  1108→def split_frame_timings(text: str, positions: List[int], duration: float, words: List[tuple]) -> dict:
  1109→  1109→    """Given a segment's full narration text, the frame positions it spans,
  1110→  1110→    the segment's total audio duration, and (optionally) real word-level
  1111→  1111→    timestamps for that audio, return {position: (start, end)} clip-local
  1112→  1112→    timing for each frame so the image swap lines up with what's actually
  1113→  1113→    being spoken. Falls back to a proportional word-count split if
  1114→  1114→    transcription isn't available."""
  1115→  1115→    frame_texts = split_into_segments(text, len(positions))
  1116→  1116→    word_counts = [len(t.split()) for t in frame_texts]
  1117→  1117→    total_words = sum(word_counts)
  1118→  1118→
  1119→  1119→    timings: dict = {}
  1120→  1120→
  1121→  1121→    if words and total_words > 0 and len(words) >= total_words:
  1122→  1122→        # Walk the real transcribed words in order, assigning the same count
  1123→  1123→        # to each frame that split_into_segments gave it textually.
  1124→  1124→        cursor = 0
  1125→  1125→        prev_end = 0.0
  1126→  1126→        for pos, count in zip(positions, word_counts):
  1127→  1127→            if count <= 0:
  1128→  1128→                timings[pos] = (prev_end, prev_end)
  1129→  1129→                continue
  1130→  1130→            chunk = words[cursor: cursor + count]
  1131→  1131→            cursor += count
  1132→  1132→            start = prev_end if pos == positions[0] else chunk[0][1]
  1133→  1133→            end = chunk[-1][2]
  1134→  1134→            timings[pos] = (start, end)
  1135→  1135→            prev_end = end
  1136→  1136→        # Make sure the final frame's end reaches the true clip end so audio
  1137→  1137→        # and video durations line up exactly (captures trailing breath/pause).
  1138→  1138→        last_pos = positions[-1]
  1139→  1139→        s, _ = timings[last_pos]
  1140→  1140→        timings[last_pos] = (s, duration)
  1141→  1141→    else:
  1142→  1142→        # Fallback: proportional split by word count (or even split if no words at all).
  1143→  1143→        cursor = 0.0
  1144→  1144→        if total_words > 0:
  1145→  1145→            for pos, count in zip(positions, word_counts):
  1146→  1146→                span = duration * (count / total_words) if total_words else 0.0
  1147→  1147→                timings[pos] = (cursor, cursor + span)
  1148→  1148→                cursor += span
  1149→  1149→        else:
  1150→  1150→            span = duration / max(1, len(positions))
  1151→  1151→            for pos in positions:
  1152→  1152→                timings[pos] = (cursor, cursor + span)
  1153→  1153→                cursor += span
  1154→  1154→        last_pos = positions[-1]
  1155→  1155→        s, _ = timings[last_pos]
  1156→  1156→        timings[last_pos] = (s, duration)
  1157→  1157→
  1158→  1158→    return timings

  1200→  1200→    (word, start, end) tuples, grouping consecutive words into short phrases
  1201→  1201→    instead of showing one word at a time. `segment_words` is built once per
  1202→  1202→    chapter in run_pipeline from the same word timestamps already produced
  1203→  1203→    for frame-timing, so nothing is transcribed twice."""

  1231→  1231→    return out_path

  1244→  1244→    caption_path: Optional[Path],
  1245→  1245→) -> Optional[Path]:
  1246→  1246→    """Render a chapter's static panel frames + per-frame narration + burned
  1247→  1247→    captions into an MP4.
  1248→  1248→
  1249→  1249→    FAST approach: uses ffmpeg concat demuxer with image durations, so a single
  1250→  1250→    ffmpeg call creates the entire chapter video (instead of N separate processes).
  1251→  1251→    This is dramatically faster than the old per-segment approach.
  1252→  1252→
  1253→  1253→    Each frame is held on screen for exactly its own audio segment's duration
  1254→  1254→    (frame_durations), so image swaps line up precisely with the narration.
  1255→  1255→    RESUME: skip entirely if chap_XXX.mp4 already exists on disk."""
  1256→  1256→    out_path = cfg.temp_chapters_dir / f"{chapter.tag}.mp4"
  1257→  1257→    if out_path.exists():
  1258→  1258→        log.info("[%s] chapter video already rendered — skipping (resume)", chapter.tag)
  1259→  1259→        return out_path
  1260→  1260→
  1261→  1261→    if not frame_paths:
  1262→  1262→        log.warning("[%s] no frames to render — skipping chapter", chapter.tag)
  1263→  1263→        return None
  1264→  1264→
  1265→  1265→    # Write a concat demuxer input file with per-image durations.
  1266→  1266→    # This lets a SINGLE ffmpeg call produce the whole slideshow video.
  1267→  1267→    concat_list = cfg.temp_chapters_dir / f"{chapter.tag}_images.txt"
  1268→  1268→    min_dur = 1.0 / FPS
  1269→  1269→    with concat_list.open("w", encoding="utf-8") as f:
  1270→  1270→        for i, (fp, d) in enumerate(zip(frame_paths, frame_durations)):
  1271→  1271→            duration = max(d, min_dur)
  1272→  1272→            # Ensure the image exists (use absolute path for concat demuxer).
  1273→  1273→            abs_fp = fp.resolve().as_posix()
  1274→  1274→            f.write(f"file '{abs_fp}'\n")
  1275→  1275→            f.write(f"duration {duration:.3f}\n")
  1276→  1276→        # ffmpeg concat demuxer needs a trailing file entry (no duration).
  1277→  1277→        last_fp = frame_paths[-1].resolve().as_posix()
  1278→  1278→        f.write(f"file '{last_fp}'\n")
  1279→  1279→
  1280→  1280→    log.info("[%s] rendering %d frames with concat demuxer", chapter.tag, len(frame_paths))
  1281→  1281→
  1282→  1282→    # Single ffmpeg call: slideshow from concat demuxer.
  1283→  1283→    # Frames are already 1920x1080 from _compose_canvas, but ensure exact size.
  1284→  1284→    video_filters = f"scale={CANVAS_W}:{CANVAS_H}:force_original_aspect_ratio=decrease,pad={CANVAS_W}:{CANVAS_H}:(ow-iw)/2:(oh-ih)/2"
  1285→  1285→    silent_path = cfg.temp_chapters_dir / f"{chapter.tag}_nosubs.mp4"
  1286→  1286→    cmd = [
  1287→  1287→        "ffmpeg", "-y",
  1288→  1288→        "-f", "concat", "-safe", "0",
  1289→  1289→        "-i", str(concat_list),
  1290→  1290→        "-vf", video_filters,
  1291→  1291→        "-c:v", "libx264", "-preset", "ultrafast", "-pix_fmt", "yuv420p",
  1292→  1292→        "-r", str(FPS),
  1293→  1293→        str(silent_path),
  1294→  1294→    ]
  1295→  1295→    try:
  1296→  1296→        run_ffmpeg(cmd)
  1297→  1297→    except RuntimeError as e:
  1298→  1298→        log.error("[%s] concat render failed: %s", chapter.tag, e)
  1299→  1299→        return None
  1300→  1300→
  1301→  1301→    # Mux audio + burn captions in a single ffmpeg pass.
  1302→  1302→    if audio_path and audio_path.exists():
  1303→  1303→        cmd = [
  1304→  1304→            "ffmpeg", "-y",
  1305→  1305→            "-i", str(silent_path),
  1306→  1306→            "-i", str(audio_path),
  1307→  1307→            "-map", "0:v", "-map", "1:a",
  1308→  1308→            "-c:v", "copy",
  1309→  1309→            "-c:a", "aac", "-b:a", AUDIO_BITRATE, "-ar", str(AUDIO_SAMPLE_RATE),
  1310→  1310→            "-shortest",
  1311→  1311→        ]
  1312→  1312→        if caption_path and caption_path.exists():
  1313→  1313→            cmd[-1:-1] = ["-vf", f"ass={_ffmpeg_escape_path(caption_path)}"]
  1314→  1314→        cmd.append(str(out_path))
  1315→  1315→        run_ffmpeg(cmd)
  1316→  1316→    else:
  1317→  1317→        if caption_path and caption_path.exists():
  1318→  1318→            run_ffmpeg([
  1319→  1319→                "ffmpeg", "-y", "-i", str(silent_path),
  1320→  1320→                "-vf", f"ass={_ffmpeg_escape_path(caption_path)}",
  1321→  1321→                "-c:v", "copy",
  1322→  1322→                "-c:a", "aac", "-b:a", AUDIO_BITRATE, "-ar", str(AUDIO_SAMPLE_RATE),
  1323→  1323→                str(out_path),
  1324→  1324→            ])
  1325→  1325→        else:
  1326→  1326→            shutil.copy2(silent_path, out_path)
  1327→  1327→
  1328→  1328→    silent_path.unlink(missing_ok=True)
  1329→  1329→    concat_list.unlink(missing_ok=True)
  1330→  1330→
  1331→  1331→    log.info("[%s] chapter video rendered -> %s", chapter.tag, out_path.name)
  1332→  1332→    return out_path
  1333→  1333→

  1335→  1335→def _ffmpeg_escape_path(p: Path) -> str:
  1336→  1336→    # ffmpeg's ass filter needs escaped colons/backslashes on some platforms.
  1337→  1337→    s = str(p).replace("\\", "/")
  1338→  1338→    s = s.replace(":", r"\:")
  1339→  1339→    return s

  1346→  1346→def run_ffmpeg(cmd: List[str]) -> None:
  1347→  1347→    log.debug("Running: %s", " ".join(cmd))
  1348→  1348→    result = subprocess.run(cmd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
  1349→  1349→    if result.returncode != 0:
  1350→  1350→        log.error("ffmpeg command failed:\n%s", result.stdout)
  1351→  1351→        raise RuntimeError(f"ffmpeg failed (exit {result.returncode}): {' '.join(cmd)}")

  1354→  1354→def merge_chapters(cfg: PipelineConfig, chapter_videos: List[Path]) -> Path:
  1355→  1355→    """Zero-reencode concat of all chapter mp4s via ffmpeg's concat demuxer."""
  1356→  1356→    concat_list = cfg.work_dir / "concat_list.txt"
  1357→  1357→    with concat_list.open("w", encoding="utf-8") as f:
  1358→  1358→        for cv in chapter_videos:
  1359→  1359→            f.write(f"file '{cv.resolve().as_posix()}'\n")
  1360→  1360→
  1361→  1361→    merged_path = cfg.work_dir / "master_merged.mp4"
  1362→  1362→    if merged_path.exists():
  1363→  1363→        log.info("Merged master file already exists — skipping merge step (resume)")
  1364→  1364→        return merged_path
  1365→  1365→
  1366→  1366→    log.info("Merging %d chapter videos via ffmpeg stream copy...", len(chapter_videos))
  1367→  1367→    run_ffmpeg(
  1368→  1368→        [
  1369→  1369→            "ffmpeg", "-y", "-f", "concat", "-safe", "0",
  1370→  1370→            "-i", str(concat_list),
  1371→  1371→            "-c", "copy",
  1372→  1372→            str(merged_path),
  1373→  1373→        ]
  1374→  1374→    )
  1375→  1375→    log.info("Merge complete -> %s", merged_path.name)
  1376→  1376→    return merged_path

  1379→  1379→def overlay_bgm(cfg: PipelineConfig, merged_path: Path) -> Path:
  1380→  1380→    """Loop BGM under the narration with SIDECHAIN COMPRESSION so the music
  1381→  1381→    automatically ducks by 18dB whenever the narrator is speaking, then rises
  1382→  1382→    back up during pauses. This is the Phase 3 sound design technique used by
  1383→  1383→    professional recap channels.
  1384→  1384→
  1385→  1385→    Filter chain:
  1386→  1386→      1. BGM is looped and low-passed slightly so it sits behind the voice.
  1387→  1387→      2. sidechaincompress uses the narration track as the sidechain input:
  1388→  1388→         when voice is active, BGM is compressed (ducked); when silent, BGM
  1389→  1389→         returns to full volume.
  1390→  1390→      3. The ducked BGM is mixed with the narration at full volume.
  1391→  1391→    """
  1392→  1392→    # Ensure the output directory exists before writing.
  1393→  1393→    cfg.output_path.parent.mkdir(parents=True, exist_ok=True)
  1394→  1394→
  1395→  1395→    if not cfg.bgm_path:
  1396→  1396→        log.info("No --bgm provided — copying merged file directly to final output")
  1397→  1397→        shutil.copy2(merged_path, cfg.output_path)
  1398→  1398→        return cfg.output_path
  1399→  1399→
  1400→  1400→    log.info("Overlaying BGM with sidechain ducking (voice-activated -18dB)...")
  1401→  1401→
  1402→  1402→    # sidechaincompress parameters:
  1403→  1403→    #   threshold=-30dB (duck starts when voice exceeds this)
  1404→  1404→    #   ratio=8 (heavy duck)
  1405→  1405→    #   attack=20ms (fast duck when voice starts)
  1406→  1406→    #   release=300ms (smooth recovery when voice stops)
  1407→  1407→    #   makeup=1 (minimum — no boost on BGM; range is 1-64)
  1408→  1408→    # Use -stream_loop -1 on the BGM input (input-level looping) instead of
  1409→  1409→    # the aloop filter, which is more reliable for long videos.
  1410→  1410→    filter_complex = (
  1411→  1411→        f"[1:a]volume=0.6,"
  1412→  1412→        f"sidechaincompress=threshold=-30dB:ratio=8:attack=20:release=300:makeup=1[bgm];"
  1413→  1413→        f"[0:a][bgm]amix=inputs=2:duration=first:dropout_transition=2,"
  1414→  1414→        f"loudnorm=I={TARGET_LOUDNESS_LUFS}:TP=-1.5:LRA=11[aout]"
  1415→  1415→    )
  1416→  1416→    run_ffmpeg(
  1417→  1417→        [
  1418→  1418→            "ffmpeg", "-y",
  1419→  1419→            "-i", str(merged_path),
  1420→  1420→            "-stream_loop", "-1", "-i", str(cfg.bgm_path),
  1421→  1421→            "-filter_complex", filter_complex,
  1422→  1422→            "-map", "0:v", "-map", "[aout]",
  1423→  1423→            "-c:v", "copy",
  1424→  1424→            "-c:a", "aac", "-b:a", AUDIO_BITRATE, "-ar", str(AUDIO_SAMPLE_RATE),
  1425→  1425→            "-shortest",
  1426→  1426→            str(cfg.output_path),
  1427→  1427→        ]
  1428→  1428→    )
  1429→  1429→    log.info("Final master video with sidechain-ducked BGM written -> %s", cfg.output_path)
  1430→  1430→    return cfg.output_path
  1431→  1431→

  1449→  1449→def run_pipeline(cfg: PipelineConfig) -> None:
  1450→  1450→    start = time.time()
  1451→  1451→    cfg.ensure_dirs()
  1452→  1452→    # Ensure the output directory exists.
  1453→  1453→    cfg.output_path.parent.mkdir(parents=True, exist_ok=True)
  1454→  1454→
  1455→  1455→    chapters = discover_chapters(cfg)
  1456→  1456→    total = len(chapters)
  1457→  1457→    chapter_videos: List[Path] = []
  1458→  1458→    prev_tail = ""
  1459→  1459→
  1460→  1460→    cfg.write_progress("render", 0, total, f"Starting pipeline over {total} chapters")
  1461→  1461→
  1462→  1462→    for chapter in chapters:
  1463→  1463→        t0 = time.time()
  1464→  1464→        log.info("=== Chapter %d/%d (%s) ===", chapter.index, total, chapter.name)
  1465→  1465→        cfg.write_progress("render", chapter.index - 1, total, f"Chapter {chapter.index}/{total}: {chapter.name}")
  1466→  1466→
  1467→  1467→        existing_video = cfg.temp_chapters_dir / f"{chapter.tag}.mp4"
  1468→  1468→        if existing_video.exists():
  1469→  1469→            log.info("[%s] fully rendered already — resuming past this chapter", chapter.tag)
  1470→  1470→            chapter_videos.append(existing_video)
  1471→  1471→            # Keep narrative continuity even when resuming: pull tail from cached script.
  1472→  1472→            script_cache = cfg.temp_scripts_dir / f"{chapter.tag}.txt"
  1473→  1473→            if script_cache.exists():
  1474→  1474→                prev_tail = last_sentences(script_cache.read_text(encoding="utf-8"))
  1475→  1475→            continue
  1476→  1476→
  1477→  1477→        # Slice returns [(frame_path, source_panel_index), ...] so we can map
  1478→  1478→        # each frame to its source image's narration for perfect sync.
  1479→  1479→        # PANEL-BY-PANEL MODE: use gutter-aware slicing that detects both
  1480→  1480→        # horizontal AND vertical panel boundaries, so each manga panel
  1481→  1481→        # gets its own individual 1920x1080 frame.
  1482→  1482→        frame_data = slice_chapter_panels(cfg, chapter)
  1483→  1483→        frame_paths = [fp for fp, _ in frame_data]
  1484→  1484→        frame_sources = [si for _, si in frame_data]
  1485→  1485→
  1486→  1486→        log.info("[%s] %d frames from %d panels (panel-by-panel slicing)", chapter.tag, len(frame_paths), len(chapter.panel_paths))
  1487→  1487→        cfg.write_progress("render", chapter.index - 1, total,
  1488→  1488→                           f"Chapter {chapter.index}/{total}: sliced {len(frame_paths)} frames")
  1489→  1489→
  1490→  1490→        # Build an ordered list of (tag, text, frame_positions) "segments" —
  1491→  1491→        # each one gets exactly ONE continuous edge-tts synthesis call, then
  1492→  1492→        # real word timestamps from that single clip drive per-frame timing.
  1493→  1493→        # This is what eliminates the audible stop-after-every-word chop:
  1494→  1494→        # previously every one of these frame positions triggered its own
  1495→  1495→        # separate edge-tts call.
  1496→  1496→        segments: List[tuple] = []  # (tag, text, positions)
  1497→  1497→
  1498→  1498→        if chapter.image_narrations:
  1499→  1499→            # PER-IMAGE NARRATION MODE (preferred): each source image has its own
  1500→  1500→            # VLM-generated narration, so what you hear matches what you see.
  1501→  1501→            log.info("[%s] per-image narration mode: %d images, %d frames",
  1502→  1502→                     chapter.tag, len(chapter.panel_paths), len(frame_paths))
  1503→  1503→
  1504→  1504→            # Translate + rephrase each image's narration individually.
  1505→  1505→            panel_narrations: dict = {}  # panel_index -> narration text
  1506→  1506→            for idx, panel_path in enumerate(chapter.panel_paths):
  1507→  1507→                raw_text = chapter.image_narrations.get(panel_path.name, "")
  1508→  1508→                img_tag = f"{chapter.tag}_img{idx + 1:03d}"
  1509→  1509→                translated = translate_text(cfg, raw_text, img_tag)
  1510→  1510→                rephrased = rephrase_text(cfg, translated, img_tag, prev_tail)
  1511→  1511→                panel_narrations[idx] = rephrased
  1512→  1512→                if rephrased:
  1513→  1513→                    prev_tail = last_sentences(rephrased)
  1514→  1514→
  1515→  1515→            # Group frame positions by source panel (preserving timeline order).
  1516→  1516→            from collections import defaultdict as _dd
  1517→  1517→            panel_frame_positions = _dd(list)  # panel_index -> [frame_positions]
  1518→  1518→            for pos, si in enumerate(frame_sources):
  1519→  1519→                panel_frame_positions[si].append(pos)
  1520→  1520→
  1521→  1521→            for si, positions in panel_frame_positions.items():
  1522→  1522→                narr = panel_narrations.get(si, "")
  1523→  1523→                segments.append((f"img{si + 1:03d}", narr, positions))
  1524→  1524→        else:
  1525→  1525→            # CHAPTER-LEVEL NARRATION MODE (backward compat for old jobs without
  1526→  1526→            # narration.json): one continuous narration clip for the whole
  1527→  1527→            # chapter, sliced across all frames by real word timing.
  1528→  1528→            log.info("[%s] chapter-level narration mode (no narration.json)", chapter.tag)
  1529→  1529→            translated_text = translate_chapter_text(cfg, chapter)
  1530→  1530→            narration = rephrase_chapter(cfg, chapter, translated_text, prev_tail)
  1531→  1531→            prev_tail = last_sentences(narration) if narration else prev_tail
  1532→  1532→            segments.append(("chapter", narration, list(range(len(frame_paths)))))
  1533→  1533→
  1534→  1534→        # Synthesize one continuous clip per segment, transcribe it once, and
  1535→  1535→        # derive per-frame timing + chapter-offset caption words from it.
  1536→  1536→        frame_timing = [None] * len(frame_paths)  # position -> (start, end) clip-local to its segment
  1537→  1537→        segment_audio_paths: List[Path] = []
  1538→  1538→        segment_words: List[tuple] = []  # (word, chapter_offset_start, chapter_offset_end)
  1539→  1539→        chapter_offset = 0.0
  1540→  1540→
  1541→  1541→        for seg_idx, (tag, text, positions) in enumerate(segments):
  1542→  1542→            cfg.write_progress("render", chapter.index - 1, total,
  1543→  1543→                               f"Chapter {chapter.index}/{total}: TTS {seg_idx+1}/{len(segments)}")
  1544→  1544→            audio_path_seg = synthesize_segment_audio(cfg, chapter, tag, text)
  1545→  1545→            duration = get_audio_duration(audio_path_seg)
  1546→  1546→            words = transcribe_words(audio_path_seg) if text.strip() else []
  1547→  1547→            timings = split_frame_timings(text, positions, duration, words)
  1548→  1548→            for pos in positions:
  1549→  1549→                frame_timing[pos] = timings[pos]
  1550→  1550→            for w_text, w_start, w_end in words:
  1551→  1551→                segment_words.append((w_text, chapter_offset + w_start, chapter_offset + w_end))
  1552→  1552→            segment_audio_paths.append(audio_path_seg)
  1553→  1553→            chapter_offset += duration
  1554→  1554→
  1555→  1555→        min_dur = 1.0 / FPS
  1556→  1556→        frame_durations = [max(min_dur, end - start) for start, end in frame_timing]
  1557→  1557→
  1558→  1558→        cfg.write_progress("render", chapter.index - 1, total,
  1559→  1559→                           f"Chapter {chapter.index}/{total}: building audio track")
  1560→  1560→        audio_path = build_chapter_audio_track(cfg, chapter, segment_audio_paths) if segment_audio_paths else None
  1561→  1561→        caption_path = generate_captions(cfg, chapter, segment_words)
  1562→  1562→
  1563→  1563→        cfg.write_progress("render", chapter.index - 1, total,
  1564→  1564→                           f"Chapter {chapter.index}/{total}: rendering {len(frame_paths)} frames")
  1565→  1565→        video_path = render_chapter(cfg, chapter, frame_paths, frame_durations, audio_path, caption_path)
  1566→  1566→
  1567→  1567→        if video_path:
  1568→  1568→            chapter_videos.append(video_path)
  1569→  1569→
  1570→  1570→        log.info("[%s] done in %.1fs", chapter.tag, time.time() - t0)
  1571→  1571→        cfg.write_progress("render", chapter.index, total, f"Chapter {chapter.index}/{total} rendered")
  1572→  1572→
  1573→  1573→    if not chapter_videos:
  1574→  1574→        cfg.write_progress("render", total, total, "No chapter videos produced", status="error")
  1575→  1575→        raise RuntimeError("No chapter videos were produced — nothing to merge.")
  1576→  1576→
  1577→  1577→    cfg.write_progress("merge", total, total, "Merging chapter videos")
  1578→  1578→    merged = merge_chapters(cfg, chapter_videos)
  1579→  1579→    cfg.write_progress("bgm", total, total, "Finalizing output")
  1580→  1580→    overlay_bgm(cfg, merged)
  1581→  1581→    cleanup_temp(cfg)
  1582→  1582→
  1583→  1583→    elapsed = time.time() - start
  1584→  1584→    cfg.write_progress("done", total, total, f"Pipeline complete in {elapsed/60:.1f} min", status="done")
  1585→  1585→    log.info("PIPELINE COMPLETE in %.1f minutes. Output: %s", elapsed / 60, cfg.output_path)
  1586→  1586→

  1592→  1592→def parse_args(argv: Optional[List[str]] = None) -> PipelineConfig:
  1593→  1593→    parser = argparse.ArgumentParser(
  1594→  1594→        description="Convert local webtoon/manhwa chapter folders into a single master recap video."
  1595→  1595→    )
  1596→  1596→    parser.add_argument(
  1597→  1597→        "--input-dir", required=True, type=Path,
  1598→  1598→        help="Root dataset folder containing chapter_001/, chapter_002/, ... subfolders.",
  1599→  1599→    )
  1600→  1600→    parser.add_argument(
  1601→  1601→        "--total-chapters", type=int, default=None,
  1602→  1602→        help="Limit processing to the first N chapters (default: process all found).",
