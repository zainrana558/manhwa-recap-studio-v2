   101→/**
   102→ * Fetch the chapter list for a manga from MangaHere.
   103→ * mangaId is the MangaHere slug (e.g. "solo_leveling").
   104→ * Returns chapters oldest-first.
   105→ */
   106→export async function fetchMangaHereChapters(
   107→  mangaSlug: string,
   108→  chapterLimit: number,
   109→): Promise<
   110→  Array<{
   111→    mangadexId: string // chapter slug e.g. "c001" (kept as mangadexId for DB compat)
   112→    chapterNum: string | null
   113→    title: string | null
   114→    language: string
   115→    pageCount: number
   116→    external: boolean
   117→  }>
   118→> {
   119→  const url = `${MANGAHERE_BASE}/manga/${mangaSlug}/`
   120→  const res = await fetch(url, {
   121→    headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36' },
   122→  })
   123→  if (!res.ok) {
   124→    throw new Error(`MangaHere chapters ${res.status} for ${mangaSlug}`)
   125→  }
   126→  const html = await res.text()
   127→
   128→  // Parse chapter links: href="/manga/{slug}/c{chapter}/1.html"
   129→  const chapters: Array<{
   130→    mangadexId: string
   131→    chapterNum: string | null
   132→    title: string | null
   133→    language: string
   134→    pageCount: number
   135→    external: boolean
   136→  }> = []
   137→  const seen = new Set<string>()
   138→  const chapterRegex = new RegExp(
   139→    `href="/manga/${mangaSlug}/((?:v\\d+/)?c[0-9.]+)/1\\.html"`,
   140→    'gi',
   141→  )
   142→  let match
   143→  while ((match = chapterRegex.exec(html)) !== null) {
   144→    const chapSlug = match[1]
   145→    if (seen.has(chapSlug)) continue
   146→    seen.add(chapSlug)
   147→    const chapterNum = chapSlug.match(/c([0-9.]+)/) ? (chapSlug.match(/c([0-9.]+)/)![1].replace(/^0+/, '') || '0') : '0'
   148→    chapters.push({
   149→      mangadexId: chapSlug,
   150→      chapterNum,
   151→      title: null,
   152→      language: 'en',
   153→      pageCount: 0,
   154→      external: false,
   155→    })
   156→  }
   157→
   158→  // MangaHere returns newest-first; reverse to oldest-first.
   159→  chapters.reverse()
   160→
   161→  // Apply chapter limit (0 = all).
   162→  const limited = chapterLimit > 0 ? chapters.slice(0, chapterLimit) : chapters
   163→  return limited
   164→}

   166→/**
   167→ * Extract all image URLs for a MangaHere chapter by scraping the chapter page HTML.
   168→ *
   169→ * MangaHere loads images via obfuscated JavaScript. The image filenames are
   170→ * embedded in the HTML as pipe-separated values. We extract them and construct
   171→ * full CDN URLs.
   172→ */
   173→export async function fetchMangaHereChapterImages(
   174→  mangaSlug: string,
   175→  chapterSlug: string,
   176→): Promise<string[]> {
   177→  const url = `${MANGAHERE_BASE}/manga/${mangaSlug}/${chapterSlug}/1.html`
   178→  const res = await fetch(url, {
   179→    headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36' },
   180→  })
   181→  if (!res.ok) {
   182→    throw new Error(`MangaHere chapter ${res.status} for ${mangaSlug}/${chapterSlug}`)
   183→  }
   184→  const html = await res.text()
   185→
   186→  // Extract store ID from cover image URL: store/manga/{storeId}/cover.jpg
   187→  const storeMatch = html.match(/store\/manga\/(\d+)/)
   188→  const storeId = storeMatch?.[1]
   189→  if (!storeId) {
   190→    throw new Error(`Could not extract store ID from ${url}`)
   191→  }
   192→
   193→  // Extract chapter folder from chapter slug: "v72/c700" → "700", "c001" → "001", "c200.5" → "200.5"
   194→  const chapterMatch = chapterSlug.match(/c([0-9.]+)/)
   195→  const chapterFolder = chapterMatch
   196→    ? chapterMatch[1].padStart(3, '0')
   197→    : chapterSlug.replace(/^c/, '').padStart(3, '0')
   198→
   199→  // Extract image filenames from obfuscated JavaScript.
   200→  // Pattern: {letter}{date}_{time}_{number} e.g. h20181105_144325_927
   201→  const filenameRegex = /([a-z]\d{8}_\d{6}_[a-z0-9]+)/gi
   202→  const filenames = new Set<string>()
   203→  let m
   204→  while ((m = filenameRegex.exec(html)) !== null) {
   205→    filenames.add(m[1])
   206→  }
   207→
   208→  if (filenames.size === 0) {
   209→    throw new Error(`No image filenames found in ${url}`)
   210→  }
   211→
   212→  // Construct full CDN URLs
   213→  const imageUrls = Array.from(filenames).map(
   214→    (fn) =>
   215→      `${MANGAHERE_CDN}/store/manga/${storeId}/${chapterFolder}.0/compressed/${fn}.jpg`,
   216→  )
   217→
   218→  return imageUrls
   219→}

   419→      language: 'en',

   421→      external: false,
   422→    })

   428→export async function fetchWebtoonsChapterImages(
   429→  titleNo: number,
   430→  episodeNo: number,
   431→): Promise<string[]> {
   432→  // Find the viewer URL from the list page.
   433→  const listRes = await fetch(
   434→    `${WEBTOONS_BASE}/en/fantasy/_/list?title_no=${titleNo}`,
   435→    {
   436→      headers: {
   437→        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
   438→        'Accept-Language': 'en-US,en;q=0.9',
   439→      },
   440→      redirect: 'follow',
   441→    },
   442→  )
   443→  if (!listRes.ok) {
   444→    throw new Error(`Webtoons list ${listRes.status}`)
   445→  }
   446→  const listHtml = await listRes.text()
   447→
   448→  const viewerRegex = new RegExp(
   449→    `href="([^"]*episode_no=${episodeNo})"`,
   450→    'i',
   451→  )
   452→  const viewerMatch = listHtml.match(viewerRegex)
   453→  if (!viewerMatch) {
   454→    throw new Error(`Episode ${episodeNo} not found for title_no=${titleNo}`)
   455→  }
   456→  const viewerUrl = viewerMatch[1].startsWith('http')
   457→    ? viewerMatch[1]
   458→    : `${WEBTOONS_BASE}${viewerMatch[1]}`
   459→
   460→  const res = await fetch(viewerUrl, {
   461→    headers: {
   462→      'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
   463→      'Accept-Language': 'en-US,en;q=0.9',
   464→      Referer: `${WEBTOONS_BASE}/`,
   465→    },
   466→  })
   467→  if (!res.ok) {
   468→    throw new Error(`Webtoons viewer ${res.status}`)
   469→  }
   470→  const html = await res.text()
   471→
   472→  // Webtoons embeds image URLs in data-url attributes.
   473→  const images: string[] = []
   474→  const dataUrlRegex = /data-url="(https:\/\/webtoon-phinf\.pstatic\.net\/[^"]+)"/gi
   475→  let m
   476→  while ((m = dataUrlRegex.exec(html)) !== null) {
   477→    images.push(m[1])
   478→  }
   479→
   480→  // Fallback: try src attributes.
   481→  if (images.length === 0) {
   482→    const srcRegex = /src="(https:\/\/webtoon-phinf\.pstatic\.net\/[^"]+)"/gi
   483→    while ((m = srcRegex.exec(html)) !== null) {
   484→      images.push(m[1])
   485→    }
   486→  }
   487→
   488→  return images
   489→}

   578→/**
   579→ * Generate an English narrative summary for a chapter by sampling up to 9
   580→ * images (first 3, middle 3, last 3) and asking the VLM to describe them.
   581→ *
   582→ * Falls back to a minimal summary on any error so the pipeline never blocks.
   583→ * NOTE: This is the OLD chapter-level summary. Prefer generateImageNarrations
   584→ * for per-image narration that stays in sync with the video frames.
   585→ */
   586→export async function generateChapterSummary(
   587→  imagePaths: string[],
   588→): Promise<string> {
   589→  if (imagePaths.length === 0) {
   590→    return 'The chapter continues the story.'
   591→  }
   592→
   593→  // Pick up to 9 sample images: first 3, middle 3, last 3.
   594→  const sample = pickSampleImages(imagePaths, 9)
   595→
   596→  try {
   597→    const zai = await getZai()
   598→
   599→    // Build content array: text prompt + each sample image as a base64 data URL.
   600→    const content: Array<
   601→      | { type: 'text'; text: string }
   602→      | { type: 'image_url'; image_url: { url: string } }
   603→    > = [
   604→      {
   605→        type: 'text',
   606→        text:
   607→          'You are summarizing a manhwa/manga chapter for a recap video. ' +
   608→          'Look at these panel images (ordered from the beginning, middle, and end of the chapter) ' +
   609→          'and write a detailed ENGLISH narrative summary of what happens: the events, ' +
   610→          'character actions, key dialogue, and emotional beats. ' +
   611→          'Write 3 to 6 sentences in third person, present tense, as if narrating a story. ' +
   612→          'Do not mention chapter numbers. Do not mention that you are looking at images. ' +
   613→          'Output only the summary text.',
   614→      },
   615→    ]
   616→
   617→    for (const p of sample) {
   618→      const buf = await fs.readFile(p)
   619→      const b64 = buf.toString('base64')
   620→      const mime = p.toLowerCase().endsWith('.png') ? 'image/png' : 'image/jpeg'
   621→      content.push({
   622→        type: 'image_url',
   623→        image_url: { url: `data:${mime};base64,${b64}` },
   624→      })
   625→    }
   626→
   627→    const zaiAny = zai as {
   628→      chat: {
   629→        completions: {
   630→          createVision: (opts: {
   631→            messages: Array<{ role: string; content: typeof content }>
   632→            thinking: { type: string }
   633→          }) => Promise<{
   634→            choices?: Array<{ message?: { content?: string } }>
   635→          }>
   636→        }
   637→      }
   638→    }
   639→
   640→    const resp = await zaiAny.chat.completions.createVision({
   641→      messages: [{ role: 'user', content }],
   642→      thinking: { type: 'disabled' },
   643→    })
   644→
   645→    const text = resp?.choices?.[0]?.message?.content?.trim()
   646→    if (text && text.length > 0) {
   647→      return text
   648→    }
   649→    return 'The chapter continues the story.'
   650→  } catch (err) {
   651→    console.error('[VLM] summary generation failed:', err)
   652→    return 'The chapter continues the story.'
   653→  }
   654→}
   655→
   656→/**
   657→ * Generate per-image narrations: send each image to the VLM individually and
   658→ * get 2-4 sentences of narration describing exactly what's in that image.
   659→ * This produces perfect sync — when the video shows image N, the narration
   660→ * describes image N.
   661→ *
   662→ * Processes images with limited concurrency (3 at a time) to balance speed
   663→ * and API load. Falls back to a generic sentence per image on any error so
   664→ * the pipeline never blocks.
   665→ *
   666→ * Returns an array of { image, text } in the same order as imagePaths.
   667→ */
   668→export async function generateImageNarrations(
   669→  imagePaths: string[],
   670→  onProgress?: (done: number, total: number) => void,
   671→): Promise<Array<{ image: string; text: string }>> {
   672→  if (imagePaths.length === 0) {
   673→    return []
   674→  }
   675→
   676→  const results: Array<{ image: string; text: string }> = new Array(imagePaths.length)
   677→  const CONCURRENCY = 3
   678→
   679→  // Process in batches of CONCURRENCY.
   680→  for (let i = 0; i < imagePaths.length; i += CONCURRENCY) {
   681→    const batch = imagePaths.slice(i, i + CONCURRENCY)
   682→    const texts = await Promise.all(
   683→      batch.map(async (imgPath, batchIdx) => {
   684→        const globalIdx = i + batchIdx
   685→        try {
   686→          const text = await narrateSingleImage(imgPath)
   687→          return text
   688→        } catch (err) {
   689→          console.error(`[VLM] image ${globalIdx + 1} narration failed:`, err)
   690→          return 'The scene continues to unfold.'
   691→        }
   692→      }),
   693→    )
   694→    for (let j = 0; j < batch.length; j++) {
   695→      const globalIdx = i + j
   696→      results[globalIdx] = {
   697→        image: path.basename(imagePaths[globalIdx]),
   698→        text: texts[j],
   699→      }
   700→    }
   701→    if (onProgress) {
   702→      onProgress(Math.min(i + CONCURRENCY, imagePaths.length), imagePaths.length)
   703→    }
   704→  }
   705→
   706→  return results
   707→}

   709→/**
   710→ * Send a single image to the VLM and get back the actual dialogue/caption
   711→ * text from its speech bubbles, thought bubbles, and caption boxes —
   712→ * transcribed as-is (translated to English), not narrated or paraphrased.
   713→ */
   714→async function narrateSingleImage(imgPath: string): Promise<string> {
   715→  const zai = await getZai()
   716→
   717→  const buf = await fs.readFile(imgPath)
   718→  const b64 = buf.toString('base64')
   719→  const ext = path.extname(imgPath).toLowerCase()
   720→  const mime = ext === '.png' ? 'image/png' : ext === '.webp' ? 'image/webp' : 'image/jpeg'
   721→
   722→  const content: Array<
   723→    | { type: 'text'; text: string }
   724→    | { type: 'image_url'; image_url: { url: string } }
   725→  > = [
   726→    {
   727→      type: 'text',
   728→      text:
   729→        'You are a precise transcriber for a webtoon/manhwa panel, not a narrator. ' +
   730→        'Look at this single panel and transcribe ONLY the actual text you can see inside ' +
   731→        'speech bubbles, thought bubbles, and caption/narration boxes — in the order a reader ' +
   732→        'would naturally read them (top to bottom, left to right within the panel). ' +
   733→        'Translate it into natural English if it is not already in English, preserving the ' +
   734→        'original meaning and tone as closely as possible.\n\n' +
   735→        'Guidelines:\n' +
   736→        '1. Output the text VERBATIM (translated) — do not paraphrase, summarize, embellish, or ' +
   737→        'add descriptive narration around it. Do not invent dialogue that is not actually written.\n' +
   738→        '2. Do not describe the artwork, action, or characters\' expressions — only transcribe ' +
   739→        'written text that literally appears in the image.\n' +
   740→        '3. If multiple bubbles/boxes are present, join them in reading order as separate ' +
   741→        'sentences, preserving punctuation like "..." and "!" as written.\n' +
   742→        '4. Sound effect text (e.g. "BOOM", "CRASH") can be included briefly if it is the only ' +
   743→        'text present, otherwise skip pure onomatopoeia in favor of actual dialogue/captions.\n' +
   744→        '5. Never mention chapter numbers, page numbers, panels, or that you are looking at an image.\n' +
   745→        '6. If the panel has NO readable text at all (a purely visual/action panel with no bubbles ' +
   746→        'or captions), output nothing at all — an empty response. Do not invent narration to fill it.\n' +
   747→        '7. Output ONLY the transcribed (translated) text — no preamble, no headers, no markdown, ' +
   748→        'no notes about what you did.',
   749→    },
   750→    {
   751→      type: 'image_url',
   752→      image_url: { url: `data:${mime};base64,${b64}` },
   753→    },
   754→  ]
   755→
   756→  const zaiAny = zai as {
   757→    chat: {
   758→      completions: {
   759→        createVision: (opts: {
   760→          messages: Array<{ role: string; content: typeof content }>
   761→          thinking: { type: string }
   762→        }) => Promise<{
   763→          choices?: Array<{ message?: { content?: string } }>
   764→        }>
   765→      }
   766→    }
   767→  }
   768→
   769→  const resp = await zaiAny.chat.completions.createVision({
   770→    messages: [{ role: 'user', content }],
   771→    thinking: { type: 'disabled' },
   772→  })
   773→
   774→  const text = resp?.choices?.[0]?.message?.content?.trim()
   775→  // No filler fallback here on purpose: a panel with no bubbles/captions is a
   776→  // real "silent" panel (pure action/scenery), and the rest of the pipeline
   777→  // already handles empty narration correctly by holding the frame briefly
   778→  // with silence instead of narrating something that was never written.
   779→  return text ?? ''
   780→}

   782→/**
   783→ * Pick up to `maxCount` sample images from a sorted list: first N, middle N, last N.
   784→ */
   785→function pickSampleImages(paths: string[], maxCount: number): string[] {
   786→  if (paths.length <= maxCount) return paths.slice()
   787→  const n = Math.floor(maxCount / 3)
   788→  const first = paths.slice(0, n)
   789→  const midStart = Math.floor(paths.length / 2) - Math.floor(n / 2)
   790→  const middle = paths.slice(midStart, midStart + n)
   791→  const last = paths.slice(paths.length - n)
   792→  // Dedupe in case of overlap on small arrays.
   793→  const seen = new Set<string>()
   794→  const out: string[] = []
   795→  for (const p of [...first, ...middle, ...last]) {
   796→    if (!seen.has(p)) {
   797→      seen.add(p)
   798→      out.push(p)
   799→    }
   800→  }
   801→  return out.slice(0, maxCount)
   802→}
